local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")
local Players = game:GetService("Players")
local DataStoreService = game:GetService("DataStoreService")
local ServerScriptService = game:GetService("ServerScriptService")

local mainModuleFunctions = require(ServerStorage.MainModuleFunctions)
local terrainDecorationsManager = require(ServerStorage.TerrainDecorationsManager)
local proceduralFunctions = require(ServerStorage.ProceduralFunctions)
local playerData = require(ServerScriptService.PlayerData)
local plots = workspace.Plots
local tableOfOccupiedPlots = {}

for i=1, 4 do
	table.insert(tableOfOccupiedPlots, "Open")
end

local tableOfTPLocations = 
{
	["1"] = Vector3.new(128, 120, 128),
	["2"] = Vector3.new(128, 120, -128),
	["3"] = Vector3.new(-128, 120, -128),
	["4"] = Vector3.new(-128, 120, 128)
}

for i,v in plots:GetChildren() do
	local plotNumber = string.sub(v.Name, -1)
	v["TeleportPart"..plotNumber].Touched:Connect(function(touchedPart)
		if tableOfOccupiedPlots[tonumber(plotNumber)] == "Closed" then
			local playerCharacter = touchedPart.Parent
			if playerCharacter:FindFirstChild("Humanoid") then
				playerCharacter:PivotTo(CFrame.new(tableOfTPLocations[plotNumber]))
			end
		end
	end)
end






Players.PlayerAdded:Connect(function(player)
	local playerId = Players:GetUserIdFromNameAsync(player.Name)
	local playerNumber = Instance.new("NumberValue")
	playerNumber.Parent = player
	playerNumber.Name = "PlayerNumber"

	print(tableOfOccupiedPlots)
	for i,v in ipairs(tableOfOccupiedPlots) do
		print(i, v)
		if v == "Open" then
			playerNumber.Value = i
			tableOfOccupiedPlots[i] = "Closed"
			break
		end
	end
	
	local playerPlotNumber = playerNumber.Value
	local playerPlot = workspace.Plots["Plot"..tostring(playerPlotNumber)]
	playerPlot.Sign.Text.SurfaceGui.TextLabel.Text = player.Name.."'s Plot"

	

	mainModuleFunctions.makeMap(playerPlotNumber, {{1,1}, {1,2}, {2,1}, {2,2}, {3,3}, {4,3}})
	terrainDecorationsManager.decorateMap(playerPlotNumber)
end)



Players.PlayerRemoving:Connect(function(player)
	local playerPlotNumber = player.PlayerNumber.Value
	local playerPlot = workspace.Plots["Plot"..tostring(playerPlotNumber)]
	playerPlot.Sign.Text.SurfaceGui.TextLabel.Text = "Empty Plot"

	mainModuleFunctions.deleteMap(playerPlotNumber)

	tableOfOccupiedPlots[player.PlayerNumber.Value] = "Open" 
end)
