local buyPlotProducts = moduleData.BuyPlotProducts

local tableOfOccupiedPlots = {}




for i=1, 4 do
	table.insert(tableOfOccupiedPlots, "Open")
end


local tableOfTPLocations = 
{
	["1"] = Vector3.new(128, 140, 128),
	["2"] = Vector3.new(128, 140, -128),
	["3"] = Vector3.new(-128, 140, -128),
	["4"] = Vector3.new(-128, 140, 128)
}

for i,v in plots:GetChildren() do
	local plotNumber = string.sub(v.Name, -1)
	v["TeleportPart"..plotNumber].Touched:Connect(function(touchedPart)
		if tableOfOccupiedPlots[tonumber(plotNumber)] == "Closed" then
			local playerCharacter = touchedPart.Parent
			if playerCharacter:FindFirstChild("Humanoid") then
				playerCharacter:PivotTo(CFrame.new(tableOfTPLocations[plotNumber]))
			end
		end
	end)
end



Players.PlayerAdded:Connect(function(player)
	local userId = player.UserId
	
	for i,v in playerData do
		v[userId] = {}
	end



	local leaderstats = Instance.new("Folder")
	leaderstats.Name = "leaderstats"
	leaderstats.Parent = player

	local coins = Instance.new("NumberValue")
	coins.Name = "Money"
	coins.Value = 0
	coins.Parent = leaderstats
	

	local playerNumber = Instance.new("NumberValue")
	playerNumber.Parent = player
	playerNumber.Name = "PlotNumber"



	for i,v in ipairs(tableOfOccupiedPlots) do
		if v == "Open" then
			playerNumber.Value = i
			tableOfOccupiedPlots[i] = "Closed"
			break
		end
	end
	
	local playerPlotNumber = playerNumber.Value

	local playerPlot = workspace.Plots["Plot"..playerPlotNumber]
	playerPlot.Sign.Text.SurfaceGui.TextLabel.Text = player.Name.."'s Plot"

	player.CharacterAdded:Connect(function(character)
		local tool = game.ReplicatedStorage:WaitForChild("MyTool"):Clone()
		tool:SetPrimaryPartCFrame(character:GetPivot())

		tool.Parent = player:WaitForChild("Backpack")
	end)

	
	local ownedChunks = playerData["OwnedChunks"][userId]
	for i = 1, 25 do
		for x = 1, 25 do
			table.insert(ownedChunks, {i,x})
		end
	end

	remoteEvents.UpdateClientData:FireClient(player, userId, "OwnedChunks", ownedChunks)
	
	SF.makeMap(playerPlotNumber, ownedChunks, player)
end)



Players.PlayerRemoving:Connect(function(player)
	local userId = player.UserId
	local plotNumber = player.PlotNumber.Value
	local plotFolder = workspace.Plots["Plot"..plotNumber]
	
	for i,v in playerData do
		v[userId] = nil
	end
	
	plotFolder.Sign.Text.SurfaceGui.TextLabel.Text = "Empty Plot"

	SF.deleteMap(plotFolder)

	tableOfOccupiedPlots[player.PlotNumber.Value] = "Open" 
	playerData[player.UserId] = nil
end)

end)

remoteEvents.CurrentBuyChunk.OnServerEvent:Connect(function(player, buyChunk)

MarketplaceService.ProcessReceipt = function(receiptInfo)
	print(receiptInfo)
	local userId = receiptInfo.PlayerId
	local player = Players:GetPlayerByUserId(userId)

	if not player then
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	local productId = receiptInfo.ProductId

	for i,v in buyPlotProducts do
		if productId == nil then
			warn("nil")
		elseif productId == v[1] then
			local coordinateString = playerData["CurrentBuyChunk"][receiptInfo.PlayerId]

			if coordinateString then
				local splitTable = string.split(coordinateString, ",")
				local coord1 = tonumber(splitTable[1])
				local coord2 = tonumber(splitTable[2])
				local plotNumber = tonumber(splitTable[3])

				local buyPlotNumber = math.max(coord1, coord2)
				
				local price
				if buyPlotNumber > 21 then
					price  = 250000
				else
					price = buyPlotProducts[buyPlotNumber][1]
				end


				--if price == receiptInfo.RobuxSpent then

					local biomes = playerData["BiomeSeeds"][userId]
					local heightSeed = playerData["HeightSeed"][userId][1]

					SF.AddChunk(coord1, coord2, plotNumber, heightSeed, importantValues.HeightScale.Value, player)
				--end
			end
			return Enum.ProductPurchaseDecision.PurchaseGranted
		end
	end



	return Enum.ProductPurchaseDecision.NotProcessedYet
end






























while true do
	for plotNumber,v in ipairs(tableOfOccupiedPlots) do
		if v == "Closed" then

			local plotFolder = plots["Plot"..plotNumber]
			local tableOfChunks = plotFolder.MainParts:GetChildren()
			local tableOfDensities = {}
			local maxObjects = 0
			
			for x,y in ipairs(tableOfChunks) do
				local biomeType = y:GetTags()[1]

				if biomeType then
					local objectDensity = moduleData.BiomeObjectDensity[biomeType]
					table.insert(tableOfDensities, objectDensity)
					maxObjects += objectDensity
				end
			end

			local currentObjects = plotFolder.NumberOfObjects.Value

			local fillRatio = currentObjects/maxObjects

			if math.random() >= fillRatio then
				local RNG = math.random(1, maxObjects)
				local chunkPos = SF.findListPos(RNG, 1, tableOfDensities)
				local chunk = tableOfChunks[chunkPos]

				terrainDecorationsManager.decorateMap(chunk, plotFolder)
				plotFolder.NumberOfObjects.Value += 1
			end
			
			
			
		end
	end
	task.wait()
end
