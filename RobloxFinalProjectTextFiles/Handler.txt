coinsBuyChunk.OnServerEvent:Connect(function(player, coordinateString)
	local splitCoords = string.split(coordinateString, ",")
	local coordsOnly = splitCoords[1]..","..splitCoords[2]
	local chunkPrice = playerData["ChunkCoinPrices"][player.UserId][coordsOnly]
	if player.leaderstats.Coins.Value >= chunkPrice then
		player.leaderstats.Coins.Value -= chunkPrice
	end
	SF.buyChunk(coordinateString, player.UserId, player)
end)




outerChunkPrice.OnServerInvoke = function(player, coordsTable)
	local priceTable = {}
	for i,v in ipairs(coordsTable) do
		local price
		local stringCoord = v[1]..","..v[2]
		local buyPlotNumber = math.max(v[1], v[2])
		
		local checkPrice = playerData["ChunkCoinPrices"][player.UserId][stringCoord]
		if checkPrice then
			price = checkPrice
		else
			price = math.round(RF.priceCacher(buyPlotNumber) * RF.getRandomFloat(0.5, 1.5))
			playerData["ChunkCoinPrices"][player.UserId][stringCoord] = price
		end
		
		table.insert(priceTable, price)
	end
	return priceTable
end

MarketplaceService.ProcessReceipt = function(receiptInfo)
	print(receiptInfo)
	local userId = receiptInfo.PlayerId
	local player = Players:GetPlayerByUserId(userId)

	if not player then
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	local productId = receiptInfo.ProductId

	for i,v in buyPlotProducts do
		if productId == nil then
			warn("nil")
				--and receiptInfo.RobuxSpent == v[2]
		elseif productId == v[1] then
			local coordinateString = playerData["CurrentBuyChunk"][receiptInfo.PlayerId]
			
			SF.buyChunk(coordinateString, userId, player, receiptInfo)
			return Enum.ProductPurchaseDecision.PurchaseGranted
		end
	end



	return Enum.ProductPurchaseDecision.NotProcessedYet
end

































while true do
	for plotNumber,v in ipairs(tableOfOccupiedPlots) do
		if v == "Closed" then
			local plotFolder = plots["Plot"..plotNumber]
			local tableOfChunks = plotFolder.ChunkFolder:GetChildren()
			local tableOfDensities = {}
			local maxObjects = 0
			
			for x,y in ipairs(tableOfChunks) do
				local biomeType = y:GetTags()[1]

				if biomeType then
					local objectDensity = moduleData.BiomeObjectDensity[biomeType]
					table.insert(tableOfDensities, objectDensity)
					maxObjects += objectDensity
				end
			end

			local currentObjects = plotFolder.NumberOfObjects.Value

			local fillRatio = currentObjects/maxObjects

			if math.random() >= fillRatio then
				local RNG = math.random(1, maxObjects)
				local chunkPos = SF.findListPos(RNG, 1, tableOfDensities)
				local chunk = tableOfChunks[chunkPos]

				terrainDecorationsManager.decorateMap(chunk, plotFolder)
				plotFolder.NumberOfObjects.Value += 1
			end
			
			
			
		end
	end
	task.wait(1)
end
