local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local CollectionService = game:GetService("CollectionService")
local UserInputService = game:GetService("UserInputService")
local TweenService = game:GetService("TweenService")
local StarterGui = game:GetService("StarterGui")


local RF = require(ReplicatedStorage:WaitForChild("ReplicatedFunctions"))
local moduleData = require(ReplicatedStorage.ModuleData)
local playerData = require(ReplicatedStorage.PlayerData)

local remoteEvents = ReplicatedStorage:WaitForChild("RemoteEvents")
local outerChunkPrice = remoteEvents:WaitForChild("OuterChunkPrice")

local chunkSizes = playerData["ChunkSizes"]
local localPlayer = Players.LocalPlayer
local playerGui = localPlayer:WaitForChild("PlayerGui")
local mouse = localPlayer:GetMouse()
local plotNumber = localPlayer:WaitForChild("PlotNumber").Value
local plots = workspace:WaitForChild("Plots")
local currentPlot = plots:WaitForChild("Plot"..plotNumber)
local mainParts = currentPlot:WaitForChild("MainParts")
local buyUIFolder = currentPlot:WaitForChild("BuyUIFolder")
local outlineBlock = ReplicatedStorage:WaitForChild("OutlineBlock")
local buyPlotUI = StarterGui:WaitForChild("BuyPlotUI")
local userId = localPlayer.UserId
local importantValues = ReplicatedStorage:WaitForChild("ImportantValues")
local priceCache = moduleData["PriceCache"]



local module = {}







function module.displayBuyChunks()
	for i,v in localPlayer:WaitForChild("PlayerGui"):WaitForChild("BillboardGuiStorage"):GetChildren() do
		v:Destroy()
	end

	local coordinatesTable = playerData["OwnedChunks"][userId]
	local buyCoordinates = RF.getBuyCoordinates(coordinatesTable)


	for i,v in buyUIFolder:GetChildren() do
		v:Destroy()
	end

	module.putIn3dSpace(buyCoordinates)
end




function module.putIn3dSpace(buyCoordinates)
	local priceTable = outerChunkPrice:InvokeServer(buyCoordinates)

	for i,v in ipairs(buyCoordinates) do
		local increment = importantValues:WaitForChild("Increment").Value
		local x, z = RF.coordinateScale(increment, plotNumber, v)



		local ySize, yPosition = module.checkNearbyY(v, plotNumber)



		local buyPlotNumber = math.max(v[1], v[2])
		if buyPlotNumber > 21 then
			buyPlotNumber = 21
		end


		local outlineBlockClone = outlineBlock:Clone()
		outlineBlockClone.Parent = buyUIFolder
		local coordSize = chunkSizes[userId][v[1]..","..v[2]]
		outlineBlockClone.Size = Vector3.new(coordSize[1], ySize, coordSize[2])
		outlineBlockClone.Position = Vector3.new(x, yPosition, z)
		outlineBlockClone.Name = v[1]..","..v[2]..","..plotNumber


		local buyPlotUIClone = buyPlotUI:Clone()
		buyPlotUIClone.Name = v[1]..","..v[2]..","..plotNumber
		buyPlotUIClone.Parent = playerGui:WaitForChild("BillboardGuiStorage")
		buyPlotUIClone.Adornee = outlineBlockClone
		buyPlotUIClone.Enabled = true
		buyPlotUIClone.StudsOffsetWorldSpace = Vector3.new(0, 3*outlineBlockClone.Size.Y/5, 0)

		local buyPlotNumber = math.max(v[1], v[2])


		
		buyPlotUIClone:WaitForChild("PriceFrame"):WaitForChild("PriceText").Text = "$ "..RF.addCommas(priceTable[i])
		
		if buyPlotNumber > 21 then
			buyPlotNumber = 21
		end
		buyPlotUIClone:SetAttribute("BuyPlotNumber", buyPlotNumber)
		local price = moduleData.BuyPlotProducts[buyPlotNumber][2]
		buyPlotUIClone:WaitForChild("RobuxBuyFrame"):WaitForChild("Frame"):WaitForChild('TextButton').Text = price
	end
end



function module.checkNearbyY(location, plotNumber)
	local ySize = 0
	local yPosition = 0
	local amountNearby = 0

	local nearbyLocations = 
	{
		{location[1] + 1, location[2]},
		{location[1], location[2] + 1},
		{location[1] - 1, location[2]},
		{location[1], location[2] - 1}
	}

	for i,v in nearbyLocations do
		local chunk = mainParts:FindFirstChild(v[1]..","..v[2]..","..plotNumber)

		if chunk then
			amountNearby += 1
			ySize += chunk.Size.Y
			yPosition += chunk.Position.Y
		end
	end

	return ySize/amountNearby, yPosition/amountNearby
end









return module
