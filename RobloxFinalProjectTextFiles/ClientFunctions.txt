local RF = require(ReplicatedStorage:WaitForChild("ReplicatedFunctions"))
local moduleData = require(ReplicatedStorage.ModuleData)
local playerData = require(ReplicatedStorage.PlayerData)

local remoteEvents = ReplicatedStorage:WaitForChild("RemoteEvents")
local outerChunkPrice = remoteEvents:WaitForChild("OuterChunkPrice")
local coinsBuyChunk = remoteEvents:WaitForChild("CoinsBuyChunk")


local localPlayer = Players.LocalPlayer
local userId = localPlayer.UserId
local playerGui = localPlayer:WaitForChild("PlayerGui")
local mouse = localPlayer:GetMouse()
local plotNumber = localPlayer:WaitForChild("PlotNumber").Value

local plots = workspace:WaitForChild("Plots")
local currentPlot = plots:WaitForChild("Plot"..plotNumber)
local chunkParts = currentPlot:WaitForChild("ChunkFolder")
local buyUIFolder = currentPlot:WaitForChild("BuyUIFolder")
local outlineBlock = ReplicatedStorage:WaitForChild("OutlineBlock")
local importantValues = ReplicatedStorage:WaitForChild("ImportantValues")
local priceCache = moduleData["PriceCache"]
local chunkSizes = playerData["ChunkSizes"]

local buyChunkBillboard = playerGui:WaitForChild("BuyChunkBillboard")
local screenGui = playerGui:WaitForChild("ScreenGui")
local buyChunkFrame = screenGui:WaitForChild("BuyChunkFrame")
local moneyText = buyChunkFrame:WaitForChild("MoneyText")

local module = {}



function module.displayBuyChunks()

	local coordinatesTable = playerData["OwnedChunks"][userId]
	local buyCoordinates = RF.getBuyCoordinates(coordinatesTable)
	local priceTable = outerChunkPrice:InvokeServer(buyCoordinates)

	for i,coord in ipairs(buyCoordinates) do
		if buyUIFolder:FindFirstChild(coord[1]..","..coord[2]..","..plotNumber) == nil then
			module.putIn3dSpace(coord, priceTable, i)
		end
	end
end




function module.putIn3dSpace(coord, priceTable, i)
	local increment = importantValues:WaitForChild("Increment").Value
	local x, z = RF.coordinateScale(increment, plotNumber, coord)
	local coordName = coord[1]..","..coord[2]..","..plotNumber

	local buyPlotNumber = math.max(coord[1], coord[2])
	if buyPlotNumber > 21 then
		buyPlotNumber = 21
	end


	local outlineBlockClone = outlineBlock:Clone()
	outlineBlockClone.Parent = buyUIFolder
	print(chunkSizes)
	local coordSize = chunkSizes[userId][coord[1]..","..coord[2]]
	outlineBlockClone.Size = Vector3.new(coordSize[1], coordSize[2], coordSize[3])
	outlineBlockClone.Position = Vector3.new(x, coordSize[2]/2, z)
	outlineBlockClone.Name = coordName


	local buyChunkBillboardClone = buyChunkBillboard:Clone()
	buyChunkBillboardClone.Name = coordName
	buyChunkBillboardClone.Parent = playerGui:WaitForChild("BuyChunkGuiStorage")
	buyChunkBillboardClone.Adornee = outlineBlockClone
	buyChunkBillboardClone.Enabled = true
	buyChunkBillboardClone.StudsOffsetWorldSpace = Vector3.new(0, 3*outlineBlockClone.Size.Y/5, 0)


	buyChunkBillboardClone:WaitForChild("PriceFrame"):WaitForChild("PriceText").Text = "$ "..RF.addCommas(priceTable[i])

	buyChunkBillboardClone:SetAttribute("BuyPlotNumber", buyPlotNumber)
	local price = moduleData.BuyPlotProducts[buyPlotNumber][2]
	buyChunkBillboardClone:WaitForChild("RobuxBuyFrame"):WaitForChild("Frame"):WaitForChild('TextButton').Text = price
end



function module.buyChunkScreenUI(coords, price)
	moneyText.Text = price
	module.tweenUIPos(buyChunkFrame, {0.5,0}, {0.5,0})
	buyChunkFrame:WaitForChild("Coordinate").Value = coords
end

function module.tweenUIPos(UI, pos1, pos2)
	local tweenInfo = TweenInfo.new(0.25, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out, 0, false, 0)
	local goal = {Position = UDim2.new(pos1[1], pos1[2], pos2[1], pos2[2])}
	local tween = TweenService:Create(UI, tweenInfo, goal)
	tween:Play()
end
