-- Services
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local CollectionService = game:GetService("CollectionService")
local TweenService = game:GetService("TweenService")

local moduleData = require(ReplicatedStorage.ModuleData)
local priceCache = moduleData.PriceCache

local module = {}

-- Gets a random decimal number from a given range
function module.getRandomFloat(min, max)
	return min + (math.random() * (max - min))
end

-- Checks if the parent or 2nd parent is a breakable part
function module.checkBreakableHierarchy(part)
	if part then
		if CollectionService:HasTag(part.Parent, "BreakableModel") then
			return part.Parent
		elseif CollectionService:HasTag(part.Parent.Parent, "BreakableModel") then
			return part.Parent.Parent
		else
			return false
		end
	end
end


-- Shakes the model in a random direction
function module.shakeModel(model)
	-- Saves the original scale and adds a red highlight for damage visual
	local scale = model:GetAttribute("Scale")
	local redHightlight = ReplicatedStorage:WaitForChild("RedHighlight"):Clone()
	redHightlight.Parent = model
	
	-- Scales a little bigger then back to a little smaller
	model:ScaleTo(scale * 1.05)
	task.delay(0.1, function()
		model:ScaleTo(scale)
		redHightlight:Destroy()
	end)
end

-- Takes a number and adds commas every 3 digits
function module.addCommas(number)
	local str = tostring(number)
	local result = ""
	local count = 0

	for i = #str, 1, -1 do
		result = str:sub(i,i) .. result
		count = count + 1

		-- Add comma after every 3 digits, except at the start
		if count == 3 and i > 1 then
			result = "," .. result
			count = 0
		end
	end

	return result
end

-- Changes the coordinate system based on what plot the player is on (because there is 4 plots)
function module.coordinateScale(increment, plotNumber, coords)
	local x, z
	if plotNumber == 1 then		
		x = coords[1] * increment + increment
		z = coords[2] * increment + increment
	elseif plotNumber == 2 then
		x = coords[1] * increment + increment
		z = coords[2] * -increment - increment
	elseif plotNumber == 3 then
		x = coords[1] * -increment - increment
		z = coords[2] * -increment - increment
	elseif plotNumber == 4 then
		x = coords[1] * -increment - increment
		z = coords[2] * increment + increment
	end
	return x, z
end

-- Saves the prices so the computer doesnt have to calculate the price everytime for buying plots
function module.priceCacher(buyPlotNumber)
	local price
	if priceCache[buyPlotNumber] then
		return priceCache[buyPlotNumber]
	else
		price = buyPlotNumber ^ 6
		priceCache[buyPlotNumber] = price
		return price
	end
end


-- Gets the buy chunk coordinates from the currently owned coordinates
function module.getBuyCoordinates(coordinatesTable)
	local buyCoordinates = {}
	for i,v in coordinatesTable do
		local downOffset = 1
		local leftOffset = 1
		
		-- Stops player from expanding in all directions and going to other plots
		if v[1] == 1 then
			downOffset = 0 
		end
		if v[2] == 1 then
			leftOffset = 0
		end

		-- Offsets cords in the 4 directions
		local offsetCoords = 
			{
				{v[1] + 1, v[2]},
				{v[1], v[2] + 1},
				{v[1] - downOffset, v[2]},
				{v[1], v[2] - leftOffset}
			}

		-- Compares all the offsetCoords and validates it with checkCoordinates function
		for i,v in offsetCoords do
			if module.checkCoordinates(v, coordinatesTable, buyCoordinates) then
				table.insert(buyCoordinates, v)
			end
		end
	end
	return buyCoordinates
end

-- Checks if the coordinate is already in the table
-- Also checks if the coordinate is an already existing chunk
function module.checkCoordinates(coord, coordinatesTable, buyCoordinates)
	for i,v in coordinatesTable do
		if coord[1] == v[1] and coord[2] == v[2] then
			return false
		end
	end
	for i,v in buyCoordinates do
		if coord[1] == v[1] and coord[2] == v[2] then
			return false
		end
	end
	return true
end

-- Replaces underscores with spaces
function module.ReplaceUnderscoresWithSpaces(input)
	return string.gsub(input, "_", " ")
end

return module
