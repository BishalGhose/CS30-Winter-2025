local priceCache = moduleData.PriceCache

local module = {}

function module.getRandomFloat(min, max)
	return min + (math.random() * (max - min))
end

function module.checkBreakableHierarchy(part)
	if part then
		if CollectionService:HasTag(part.Parent, "BreakableModel") then
			return part.Parent
		elseif CollectionService:HasTag(part.Parent.Parent, "BreakableModel") then
			return part.Parent.Parent
		else
			return false
		end
	end
end


function module.shakeModel(model)
	local originalCFrame = model:GetPivot()
	local scale = model:GetAttribute("Scale")

	model:ScaleTo(scale * 1.05)
	task.wait(0.1)
	model:ScaleTo(scale)
end

function module.addCommas(num)
	local str = tostring(num)
	local result = ""
	local count = 0

	for i = #str, 1, -1 do
		result = str:sub(i,i) .. result
		count = count + 1
		if count == 3 and i > 1 then
			result = "," .. result
			count = 0
		end
	end

	return result
end

function module.coordinateScale(increment, plotNumber, coords)
	local x, z
	if plotNumber == 1 then		
		x = coords[1] * increment + increment
		z = coords[2] * increment + increment
	elseif plotNumber == 2 then
		x = coords[1] * increment + increment
		z = coords[2] * -increment - increment
	elseif plotNumber == 3 then
		x = coords[1] * -increment - increment
		z = coords[2] * -increment - increment
	elseif plotNumber == 4 then
		x = coords[1] * -increment - increment
		z = coords[2] * increment + increment
	end
	return x, z
end


function module.priceCacher(buyPlotNumber)
	local price
	if priceCache[buyPlotNumber] then
		return priceCache[buyPlotNumber]
	else
		price = buyPlotNumber ^ 6
		priceCache[buyPlotNumber] = price
		return price
	end
end

function module.getBuyCoordinates(coordinatesTable)
	local buyCoordinates = {}
	for i,v in coordinatesTable do
		local downOffset = 1
		local leftOffset = 1
		if v[1] == 1 then
			downOffset = 0 
		end
		if v[2] == 1 then
			leftOffset = 0
		end

		local offsetCoords = 
			{
				{v[1] + 1, v[2]},
				{v[1], v[2] + 1},
				{v[1] - downOffset, v[2]},
				{v[1], v[2] - leftOffset}
			}

		for i,v in offsetCoords do
			if module.checkCoordinates(v, coordinatesTable, buyCoordinates) then
				table.insert(buyCoordinates, v)
			end
		end
	end
	return buyCoordinates
end


function module.checkCoordinates(coord, coordinatesTable, buyCoordinates)
	for i,v in coordinatesTable do
		if coord[1] == v[1] and coord[2] == v[2] then
			return false
		end
	end
	for i,v in buyCoordinates do
		if coord[1] == v[1] and coord[2] == v[2] then
			return false
		end
	end
	return true
end

return module
