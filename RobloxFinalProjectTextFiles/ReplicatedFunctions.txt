local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local CollectionService = game:GetService("CollectionService")
local TweenService = game:GetService("TweenService")
	


local module = {}

function module.getRandomFloat(min, max)
	return min + (math.random() * (max - min))
end

function module.checkBreakableHierarchy(part)
	if part then
		if CollectionService:HasTag(part.Parent, "BreakableModel") then
			return part.Parent
		elseif CollectionService:HasTag(part.Parent.Parent, "BreakableModel") then
			return part.Parent.Parent
		else
			return false
		end
	end
end


function module.shakeModel(model)
	local originalCFrame = model:GetPivot()

	local randomDirection = Vector3.new(
		math.random(-100, 100) / 100,
		math.random(-100, 100) / 100,
		math.random(-100, 100) / 100
	).Unit

	local offset = randomDirection * 0.2 


	local tableOfOffsets = {offset, - offset, Vector3.zero}

	for i,v in tableOfOffsets do
		model:PivotTo(originalCFrame + v)
		task.wait(0.1)
	end
end


function module.breakModel(model)
	for i = 1, 10 do
		for i,v in model:GetDescendants() do
			if v:IsA("Part") then
				v.Transparency += 0.1
				task.wait()
			end
		end
	end

	model:Destroy()
end




function module.coordinateScale(increment, plotNumber, coords)
	local x, z
	if plotNumber == 1 then		
		x = coords[1] * increment + increment
		z = coords[2] * increment + increment
	elseif plotNumber == 2 then
		x = coords[1] * increment + increment
		z = coords[2] * -increment - increment
	elseif plotNumber == 3 then
		x = coords[1] * -increment - increment
		z = coords[2] * -increment - increment
	elseif plotNumber == 4 then
		x = coords[1] * -increment - increment
		z = coords[2] * increment + increment
	end
	return x, z
end




return module
