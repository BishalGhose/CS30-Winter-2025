local ReplicatedStorage = game:GetService("ReplicatedStorage")
local CollectionService = game:GetService("CollectionService")
local RunService = game:GetService("RunService")
local module = {}


function module.ReplaceBlock(part, GrassType)
	for i,v in part:GetChildren() do
		v:Destroy()
	end
	
	local offsetColor1 = math.random(-10,10)/255
	local red = math.clamp(GrassType.Dirt.Color.R + offsetColor1, 0, 1)
	local green = GrassType.Dirt.Color.G
	local blue = math.clamp(GrassType.Dirt.Color.B + offsetColor1, 0, 1)
	part.Color = Color3.new(red, green, blue)
	part.Material = GrassType.Dirt.Material
	
	local GrassClone = GrassType.Grass:Clone()
	GrassClone.Parent = part
	GrassClone.Size = Vector3.new(part.Size.X, 1, part.Size.Z)
	GrassClone.Position = Vector3.new(part.Position.X, part.Position.Y + part.Size.Y/2 + 0.5, part.Position.Z)

	if GrassClone:FindFirstChild("Texture") then
		local Texture = GrassClone:FindFirstChild("Texture")
		local offsetColor2 = math.random(-15,15)/255

		local red = math.clamp(Texture.Color3.R + offsetColor2, 0, 1)
		local green = math.clamp(Texture.Color3.G + offsetColor2, 0, 1)
		local blue = math.clamp(Texture.Color3.B + offsetColor2, 0, 1)
		
		for i,v in GrassClone:GetChildren() do
			v.Color3 = Color3.new(red, green, blue)
		end
	else
		local offsetColor2 = math.random(-15,15)/255
		
		local red = math.clamp(GrassClone.Color.R + offsetColor2, 0, 1)
		local green = math.clamp(GrassClone.Color.G + offsetColor2, 0, 1)
		local blue = math.clamp(GrassClone.Color.B + offsetColor2, 0, 1)
		GrassClone.Color = Color3.new(red, green, blue)
	end

end

function module.MakeBiomeNoise(x, z, scale, increment, name, seed, part)

	if CollectionService:HasTag(part, "WaterBlock")  then
		return
	end
	
	
	if name == "NormalGrass" then
		local noiseLocation = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)
		local Biomes = game.ReplicatedStorage.Biomes

		if noiseLocation > -0.2 then
			module.ReplaceBlock(part, Biomes.NormalGrass)
		else
			module.ReplaceBlock(part, Biomes.ForestGrass)
		end
	else
		local noiseLocation = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)
		if noiseLocation < -0.2 then
			module.ReplaceBlock(part, game.ReplicatedStorage.Biomes[name])
		end
	end
end



function module.HeightNoise(x, z, size, scale, increment, seed1, part)

	
	local noiseLocation = math.noise((x + seed1) / (increment/2.5) * scale, (z + seed1) / (increment/2.5) * scale)
	local normalized = noiseLocation + 1           
	local y = normalized * 100

	local middleValue = size/2

	local center = Vector2.new(size/2, size/2)
	local pos = Vector2.new(x, z)

	local distFromCenter = (pos - center).Magnitude
	local maxDist = (Vector2.new(0, 0) - center).Magnitude
	local invertedDist = 1 - (distFromCenter / maxDist)
	local yExtraSize = invertedDist * 500



	if noiseLocation > 0.5 then
		part.Size = Vector3.new(increment + math.random(increment/2), y + yExtraSize, increment + math.random(increment/2))
	else
		part.Size = Vector3.new(increment + math.random(increment*2), y + yExtraSize, increment + math.random(increment*2))
	end

	if noiseLocation < -0.3 then
		CollectionService:AddTag(part, "WaterBlock")
		part.Color = Color3.fromRGB(0,0,255)
	end



	part.Anchored = true
	part.Position = Vector3.new(x, y/2 - yExtraSize/2, z)
	part.Parent = workspace.MainParts
end
	
function module.MakeMap()
	local ModuleData = require(ReplicatedStorage.ModuleData)
	
	local size = 1536
	local scale = 0.02
	local increment = 64
	local seed1 = math.random(1,100000000)
	local seed2 = math.random(1,100000000)
	
	local listOfValidBiomes = {
		{"NormalGrass", math.random(1,100000000)}
	}
	
	for i,v in ModuleData.BiomeChanceData do
		if math.random(1,v) == v then
			table.insert(listOfValidBiomes, {i, math.random(1,100000000)})
		end
	end
	
		
	
	for x = increment, size, increment do
		for z = increment, size, increment do
			local part = ReplicatedStorage.NormalPart:Clone()

			module.HeightNoise(x, z, size, scale, increment, seed1, part)
			

			for i,v in listOfValidBiomes do
				module.MakeBiomeNoise(x, z, scale, increment, v[1], v[2], part)
			end
		end
	end
	task.wait(0.05)
end
