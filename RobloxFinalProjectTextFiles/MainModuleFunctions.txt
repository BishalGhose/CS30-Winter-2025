local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")
local CollectionService = game:GetService("CollectionService")
local RunService = game:GetService("RunService")

local CF = require(ServerStorage.CommonFunctions)
local moduleData = require(ServerStorage.ModuleData)

local module = {}


function module.replaceBlock(part, GrassType)
	for i,v in part:GetChildren() do
		v:Destroy()
	end

	for i,v in CollectionService:GetTags(part) do
		CollectionService:RemoveTag(part, v)
	end

	CollectionService:AddTag(part, GrassType.Name)

	
	local offsetColor1 = math.random(-10,10)/255
	local red = math.clamp(GrassType.Dirt.Color.R + offsetColor1, 0, 1)
	local green = GrassType.Dirt.Color.G
	local blue = math.clamp(GrassType.Dirt.Color.B + offsetColor1, 0, 1)
	part.Color = Color3.new(red, green, blue)
	part.Material = GrassType.Dirt.Material
	
	local GrassClone = GrassType.Grass:Clone()
	GrassClone.Parent = part
	GrassClone.Size = Vector3.new(part.Size.X, 1, part.Size.Z)
	GrassClone.Position = Vector3.new(part.Position.X, part.Position.Y + part.Size.Y/2 + CF.getRandomFloat(0.45,0.5), part.Position.Z)


	local offsetColor2 = math.random(-15,15)/255
		
	local red = math.clamp(GrassClone.Color.R + offsetColor2, 0, 1)
	local green = math.clamp(GrassClone.Color.G + offsetColor2, 0, 1)
	local blue = math.clamp(GrassClone.Color.B + offsetColor2, 0, 1)
	GrassClone.Color = Color3.new(red, green, blue)
end





function module.makeBiomeNoise(x, z, scale, increment, data, part)
	if CollectionService:HasTag(part, "Water")  then
		return
	end
	local name = data[1]
	local chanceData = data[2]
	local seed = data[3]
	
	local noiseLocation = math.noise((x + seed) / (increment) * scale, (z + seed) / (increment) * scale)

	if noiseLocation >= chanceData[1] then
		module.replaceBlock(part, ReplicatedStorage.Biomes[name])
	end
end



function module.makeHeightNoise(x, z, scale, increment, seed1, plotNumber, part)
	
	local noiseLocation = math.noise((x + seed1) / (increment/2.5) * scale, (z + seed1) / (increment/2.5) * scale)
	local normalized = noiseLocation + 1           
	local y = normalized * 100
	
	part:SetAttribute("yValue", y)

	
	
	local yMainPartLocation = y/2

	part.Size = Vector3.new(increment + CF.getRandomFloat(0, increment), y, increment + CF.getRandomFloat(0, increment))
	part.Anchored = true
	part.Position = Vector3.new(x, yMainPartLocation, z)
	part.Parent = workspace.Plots["Plot"..tostring(plotNumber)].MainParts
	
	if noiseLocation < -0.4 then
		CollectionService:AddTag(part, "Water")
		
		part.Material = Enum.Material.Sand
		local difference1 = math.random(-15,15)
		part.Color = Color3.fromRGB(199 + difference1, 159 + difference1, 119 + difference1)

		local waterBlock = ReplicatedStorage.WaterBlock:Clone()
		waterBlock.Parent = part

		local ySize = 60 - (yMainPartLocation + part.Size.Y/2)
		waterBlock.Size = Vector3.new(part.Size.X, ySize, part.Size.Z)
		local yPosition = yMainPartLocation + part.Size.Y/2 + ySize/2
		waterBlock.Position = Vector3.new(x, yPosition, z)
		CF.colorOffsetPart(waterBlock, -15, 15)
	end
end
	
function module.makeMap(plotNumber, tableOfCoords)
	local heightScale = CF.getRandomFloat(.020, .028)
	local biomeScale = 0.0185
	local increment = 64

	
	
	local listOfBiomes = {}
	for biomeName, chanceData in moduleData.BiomeChanceData do
		table.insert(listOfBiomes, {biomeName, chanceData, math.random(1,100000000)})
	end

	local seed = math.random(1,100000000)
	while module.checkHeightSeed(plotNumber, heightScale, increment, seed) == false do
		seed = math.random(1,100000000)
	end

	for i,v in tableOfCoords do
		local part = ReplicatedStorage.ChunkBlock:Clone()
		local x, z
		if plotNumber == 1 then
			x = v[1] * increment + increment
			z = v[2] * increment + increment
		elseif plotNumber == 2 then
			x = v[1] * -increment - increment
			z = v[2] * increment + increment
		elseif plotNumber == 3 then
			x = v[1] * increment + increment
			z = v[2] * -increment - increment
		elseif plotNumber == 4 then
			x = v[1] * -increment - increment
			z = v[2] * -increment - increment
		end


		module.makeHeightNoise(x, z, heightScale, increment, seed, plotNumber, part)

		for i, data in listOfBiomes do
			module.makeBiomeNoise(x, z, biomeScale * data[2][2], increment, data, part)
		end
	end
	


	for i,v in workspace.Plots["Plot"..tostring(plotNumber)].MainParts:GetChildren() do
		
		if #v:GetTags() == 0 then
			
			module.replaceBlock(v, ReplicatedStorage.Biomes.NormalGrass)
			CollectionService:AddTag(v, "NormalGrass")
		end
	end
end


function module.deleteMap(plotNumber)
	local plotFolder = workspace.Plots["Plot"..tostring(plotNumber)]

	for i, v in plotFolder:GetChildren() do
		for x, y in v:GetChildren() do
			if y:IsA("Highlight") == false then
				y:Destroy()
			end
		end
	end
end

function module.checkHeightSeed(plotNumber, scale, increment, seed)
	local x, z
	local doubleIncrement = increment*2
	if plotNumber == 1 then
		x = doubleIncrement
		z = doubleIncrement
	elseif plotNumber == 2 then
		x = -doubleIncrement
		z = doubleIncrement
	elseif plotNumber == 3 then
		x = doubleIncrement
		z = -doubleIncrement
	elseif plotNumber == 4 then
		x = -doubleIncrement
		z = -doubleIncrement
	end
	
	local noiseLocation = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)
	if noiseLocation < -0.5 or noiseLocation > 0.5 then
		return false
	end
	return true
end


return module
