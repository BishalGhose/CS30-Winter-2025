-- Services
local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local StarterPlayer = game:GetService("StarterPlayer")
local StarterGui = game:GetService("StarterGui")

-- Player things and random things
local localPlayer = Players.LocalPlayer
local userId = localPlayer.UserId
local playerGui = localPlayer:WaitForChild("PlayerGui")
local HPBillboard = playerGui:WaitForChild("HPBillboard")
local HPGuiStorage = playerGui:WaitForChild("HPGuiStorage")
local leaderstats = localPlayer:WaitForChild("leaderstats")
local coins = leaderstats:WaitForChild("Coins")
local mouse = localPlayer:GetMouse()
local plot = workspace:WaitForChild("Plots")
local plotNumber = localPlayer:WaitForChild("PlotNumber").Value
local currentPlot = plot:WaitForChild("Plot"..plotNumber)
local buyUIFolder = currentPlot:WaitForChild("BuyUIFolder")
local soundsFolder = ReplicatedStorage:WaitForChild("SoundEffects")

-- Remote Events
local remoteEvents = ReplicatedStorage:WaitForChild("RemoteEvents")
local currentChunkRemote = remoteEvents:WaitForChild("CurrentChunkRemote")
local updateClientData = remoteEvents:WaitForChild("UpdateClientData")
local coinsBuyChunk = remoteEvents:WaitForChild("CoinsBuyChunk")
local destroySkeleton = remoteEvents:WaitForChild("DestroySkeleton")
local attackObject = remoteEvents:WaitForChild("AttackObject")

-- Module Scripts
local playerData = require(ReplicatedStorage:WaitForChild("PlayerData"))
local moduleData = require(ReplicatedStorage:WaitForChild("ModuleData"))
local RF = require(ReplicatedStorage:WaitForChild("ReplicatedFunctions"))
local CF = require(StarterPlayer.StarterPlayerScripts:WaitForChild("ClientFunctions"))

-- Destroys the skeleton that displays when mouse hovers over buy gui when remote is fired by server
destroySkeleton.OnClientEvent:Connect(function(fullCoord)
	local skeleton = buyUIFolder:FindFirstChild(fullCoord)
	if skeleton then
		skeleton:Destroy()
	end
end)


-- Updates the data on the player that the server gives
updateClientData.OnClientEvent:Connect(function(tableType, data)
	playerData[tableType][userId] = data
	if tableType == "OwnedChunks" then
		CF.displayBuyChunks() 
	end
end)


-- Destroys the ui of buy chunk, server fires this when purchase successful
currentChunkRemote.OnClientEvent:Connect(function(coordinateString)
	playerGui:WaitForChild("BuyChunkGuiStorage"):FindFirstChild(coordinateString):Destroy()
end)


-- Checks the 3d location of the mouse on where the mouse clicked, then gets the target hit
-- Checks if its a breakable object and makes a HPBar if it doesnt already exist and if the server returns continue
-- (meaning that hp is not =< 0 then we continue attacking)

mouse.Button1Down:Connect(function()
	local target = mouse.Target -- gets the Object in 3d space mouse hit
	local model = RF.checkBreakableHierarchy(target)
	if model then
		local state = attackObject:InvokeServer(model) -- Gets validation from server tto see if hp <= 0
		soundsFolder:WaitForChild("Hit"):Play()
		if state == "continue" then
			local HP = model:WaitForChild("HP").Value
			local currentHP = model:WaitForChild("CurrentHP").Value

			-- Checks if a HPBar already exists
			local HPBar 
			for i,v in HPGuiStorage:GetChildren() do
				if v.Adornee == model.PrimaryPart then
					HPBar = v
				end
			end

			-- If HPBar doesnt exist for that chunk yet then it creates one
			if HPBar == nil then
				HPBar = HPBillboard:Clone()
				HPBar.Parent = HPGuiStorage
				HPBar.Adornee = model.PrimaryPart
				HPBar.Name = model.Name
			end

			local redBar = HPBar:WaitForChild("RedBar")
			local greenBar = redBar:WaitForChild("GreenBar")
			local priceText = redBar:WaitForChild("PriceText")
			
			-- Updates the size of the ui depending on % ratio of currentHP/ HP
			if currentHP/HP >= 0 then
				priceText.Text = currentHP
				greenBar.Size = UDim2.new(currentHP/HP, 0, 1, 0)
			end

			-- Basically clones a part from replicated storage that holds the particle effect when clicking an object
			-- Then moves that object to the hitposition and plays the particle effectt
			local hitPos = mouse.Hit.Position
			local tapPart = ReplicatedStorage:WaitForChild("TapPart"):Clone()
			local particle = tapPart:WaitForChild("Attachment"):WaitForChild("ParticleEmitter")
			tapPart.Parent = workspace
			tapPart.Position = hitPos
			particle:Emit(1)
			
			-- Destroys part later so particle would be finished playing before deletion
			task.delay(3, function()
				tapPart:Destroy()
			end)
		else
			-- Destroys all the current gui objects
			for i,v in HPGuiStorage:GetChildren() do
				v:Destroy()
			end
		end
			
	end
end)
