local ReplicatedStorage = game:GetService("ReplicatedStorage")
local CollectionService = game:GetService("CollectionService")
local ServerScriptService = game:GetService("ServerScriptService")
local ServerStorage = game:GetService("ServerStorage")
local RunService = game:GetService("RunService")

local moduleData = require(ReplicatedStorage.ModuleData)
local proceduralFunctions = require(ServerScriptService.ProceduralFunctions)
local RF = require(ReplicatedStorage.ReplicatedFunctions)





local module = {}





function module.decorateMap(chunk, plotFolder)
	local biomeType = chunk:GetTags()[1]
	local RNG = RF.getRandomFloat(1,1000)
	local numberOfObjects = plotFolder.NumberOfObjects


	if biomeType == "Water" then


	elseif biomeType == "NormalGrass" then ------------------------------------------------------------------
		if RNG > 900 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", 0, 1)
		elseif RNG > 500 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		elseif RNG > 200 then
			proceduralFunctions.makeBasicBush(chunk, biomeType, plotFolder)
		elseif RNG > 150 then
			proceduralFunctions.makeOreBlock(chunk, biomeType, plotFolder)
		elseif RNG > 100 then
			proceduralFunctions.makeBerryBush(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end


	elseif biomeType == "ForestGrass" then ---------------------------------------------------
		if RNG > 980 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -2, 1)
		elseif RNG > 600 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		elseif RNG > 430 then
			proceduralFunctions.makeBamboo(chunk, biomeType, plotFolder)
		elseif RNG > 310 then
			proceduralFunctions.makeBasicBush(chunk, biomeType, plotFolder)
		elseif RNG > 280 then
			proceduralFunctions.makeBerryBush(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end


	elseif biomeType == "DesertGrass" then ------------------------------------------------------------------
		if RNG > 850 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -3, 1)
		elseif RNG > 350 then
			proceduralFunctions.makeCactus(chunk, biomeType, plotFolder)
		elseif RNG > 250 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Flat", -3, math.random(10,20)/20)
		elseif RNG > 150 then
			proceduralFunctions.makeBasicBush(chunk, biomeType, plotFolder)
		else
			proceduralFunctions.makeBerryBush(chunk, biomeType, plotFolder)
		end


	elseif biomeType == "StoneyGrass" then --------------------------------------------------------------------
		if RNG > 500 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Spikey", 0, 1)
		elseif RNG > 400 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", 0, 1)
		elseif RNG > 200 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Flat", -3, 1) 
		else
			numberOfObjects.Value -= 1
		end


	elseif biomeType == "SnowyGrass" then ------------------------------------------------------------------------------------
		if RNG > 850 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -2, 1.5)
		elseif RNG > 700 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		elseif RNG > 500 then
			proceduralFunctions.makeBasicBush(chunk, biomeType, plotFolder)
		elseif RNG > 420 then
			proceduralFunctions.makeBerryBush(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end


	elseif biomeType == "DirtGrass" then ------------------------------------------------------------------------------------
		if RNG > 600 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -2, 1)
		elseif RNG > 100 then
			proceduralFunctions.makeDeadTree(chunk, biomeType, plotFolder)
		else 
			proceduralFunctions.makeOreBlock(chunk, biomeType, plotFolder)
		end

	elseif biomeType == "AutumnGrass" then ------------------------------------------------------------------------------------
		if RNG > 930 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -2, 1)
		elseif RNG > 700 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		elseif RNG > 500 then
			proceduralFunctions.makeBasicBush(chunk, biomeType, plotFolder)
		elseif RNG > 420 then
			proceduralFunctions.makeBerryBush(chunk, biomeType, plotFolder)
		elseif RNG > 200 then
			proceduralFunctions.makeSavanahTree(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end

	elseif biomeType == "FairyGrass" then ------------------------------------------------------------------------------------
		if RNG > 950 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -2, 1)
		elseif RNG > 700 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		elseif RNG > 500 then
			proceduralFunctions.makeBasicBush(chunk, biomeType, plotFolder)
		elseif RNG > 420 then
			proceduralFunctions.makeBerryBush(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end

	elseif biomeType == "TwilightGrass" then ------------------------------------------------------------------------------------
		if RNG > 950 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -3, 1)
		elseif RNG > 700 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		elseif RNG > 500 then
			proceduralFunctions.makeBasicBush(chunk, biomeType, plotFolder)
		elseif RNG > 420 then
			proceduralFunctions.makeBerryBush(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end


	elseif biomeType == "CrimsonGrass" then ------------------------------------------------------------------------------------
		if RNG > 950 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -3, 1)
		elseif RNG > 700 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		elseif RNG > 500 then
			proceduralFunctions.makeBasicBush(chunk, biomeType, plotFolder)
		elseif RNG > 420 then
			proceduralFunctions.makeBerryBush(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end

	elseif biomeType == "HolyGrass" then ------------------------------------------------------------------------------------
		if RNG > 970 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -4, 1)
		elseif RNG > 700 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end

	elseif biomeType == "DarkDimensionGrass" then ------------------------------------------------------------------------------------
		if RNG > 900 then
			proceduralFunctions.makeRock(chunk, biomeType, plotFolder, "Normal", -4, 1)
		elseif RNG > 600 then
			proceduralFunctions.makeDefaultTree(chunk, biomeType, plotFolder)
		else
			numberOfObjects.Value -= 1
		end
	end
end




return module
