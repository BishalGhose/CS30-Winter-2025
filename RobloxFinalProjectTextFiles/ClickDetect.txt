-- Services
local MarketplaceService = game:GetService("MarketplaceService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

-- Module Scripts
local moduleData = require(ReplicatedStorage:WaitForChild("ModuleData"))

-- Random things
local localPlayer = Players.LocalPlayer
local button = script.Parent
local buyPlotUI = button.Parent.Parent.Parent
local buyPlotNumber = buyPlotUI:GetAttribute("BuyPlotNumber")
local remoteEvents = ReplicatedStorage:WaitForChild("RemoteEvents")
local currentBuyChunk = remoteEvents:WaitForChild("CurrentChunkRemote")
local clickSound = ReplicatedStorage:WaitForChild("SoundEffects"):WaitForChild("Click")

-- When button is clicked it prompts a developer product purchase, plays sound and updates the current chunk the player is 
-- Interacting with via rmeote event
button.MouseButton1Click:Connect(function()
	clickSound:Play()
	MarketplaceService:PromptProductPurchase(localPlayer, moduleData.BuyPlotProducts[buyPlotNumber][1])
	currentBuyChunk:FireServer(buyPlotUI.Adornee.Name)
end)

-- Makes the selection box/chunk skeleton visible when mouse hovers
button.MouseEnter:Connect(function()
	if buyPlotUI.Adornee then
		buyPlotUI.Adornee:FindFirstChild("SelectionBox").Visible = true
	end
end)
-- Makes the selection box/chunk skeleton invisible when mouse leaves
button.MouseLeave:Connect(function()
	if buyPlotUI.Adornee then
		buyPlotUI.Adornee:FindFirstChild("SelectionBox").Visible = false
	end
end)
