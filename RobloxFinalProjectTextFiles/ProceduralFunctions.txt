function module.makeRock(chunkPart, biome, plotFolder, shape, heightDifference, scaleFactor)
	local model = SF.makePartModel(chunkPart, shape.."_Rock")
	local grass = chunkPart.Grass      

	if shape == "Normal" then ---------------------------------------------------------------------
		
		for i = 1, math.random(2,3) do
			SF.studMaker(
				biomesFolder[biome].Stud,
				model,
				Vector3.new(RF.getRandomFloat(-7, 7), RF.getRandomFloat(-.5, .5) + heightDifference, RF.getRandomFloat(-7, 7)),
				Vector3.new(RF.getRandomFloat(1, 7), RF.getRandomFloat(1, 3), RF.getRandomFloat(1, 7)),
				-20,
				20
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		SF.finalizePosition(model, grass, 0.05, heightDifference)

	elseif shape == "Spikey" then ---------------------------------------------------------------------
		for i = 1, math.random(3,4) do
			SF.studMaker(
				biomesFolder[biome].Stud,
				model,
				Vector3.new(RF.getRandomFloat(-4, 4), RF.getRandomFloat(-1, 8), RF.getRandomFloat(-4, 4)),
				Vector3.new(RF.getRandomFloat(-3, 3), RF.getRandomFloat(15, 35), RF.getRandomFloat(-3, 3)),
				-35,
				15
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		SF.finalizePosition(model, grass, 0.04, heightDifference)

	elseif shape == "Flat" then ---------------------------------------------------------------------
		for i = 1, math.random(1,2) do
			SF.studMaker(
				biomesFolder[biome].Stud,
				model,
				Vector3.new(RF.getRandomFloat(-3, 3), RF.getRandomFloat(-2, 2), RF.getRandomFloat(-3, 3)),
				Vector3.new(RF.getRandomFloat(13, 23), RF.getRandomFloat(1.3, 2.3), RF.getRandomFloat(13, 23)),
				-35,
				15
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		SF.finalizePosition(model, grass, 0.02, heightDifference)
	end
	
	SF.runGoldenChance(model, plotFolder)
	SF.runRainbowChance(model, plotFolder)
	SF.setValueAndHP(model)
end








function module.makeCactus(chunkPart, biome, plotFolder)
	function SF.updateCactusBorders(cactusPart, offset)
		cactusPart.bordersidey.Size = Vector3.new(cactusPart.Size.X - offset, cactusPart.Size.Y + offset, cactusPart.Size.Z - offset)
		cactusPart.bordersidex.Size = Vector3.new(cactusPart.Size.X + offset, cactusPart.Size.Y - offset, cactusPart.Size.Z - offset)
		cactusPart.bordersidez.Size = Vector3.new(cactusPart.Size.X - offset, cactusPart.Size.Y - offset, cactusPart.Size.Z + offset)
	end

	local model = SF.makePartModel(chunkPart, "Cactus")

	local mainCactusPart = biomesFolder["DesertGrass"].CactusPart:Clone()
	mainCactusPart.Parent = model
	mainCactusPart.Name = "MainCactusPart"

	local cactusThickness = RF.getRandomFloat(2, 3)
	local cactusYSize = (RF.getRandomFloat(1.9, 2.8))^2 + cactusThickness*2

	mainCactusPart.Size = Vector3.new(cactusThickness, cactusYSize, cactusThickness)
	SF.updateCactusBorders(mainCactusPart, 0.5)
	
	
	SF.make4SideBranch(mainCactusPart,  biomesFolder.DesertGrass["CactusPart"], cactusThickness, "Cactus", 1, 2, -mainCactusPart.Size.Y/4)
	
	
	local grass = chunkPart.Grass
	model.PrimaryPart = mainCactusPart
	SF.finalizePosition(model, grass, 0.1, mainCactusPart.Size.Y/2)
	SF.runGoldenChance(model, plotFolder)
	SF.runRainbowChance(model, plotFolder)
end

function module.makeBamboo(chunkPart, biome, plotFolder)
	local model = SF.makePartModel(chunkPart, "Bamboo")
	

	local ySize = biomesFolder[biome].Bamboo.TallPart.Size.Y
	local numberOfParts = math.random(1,3)
	
	for i = 1 , numberOfParts do
		local bamboo = biomesFolder[biome].Bamboo:Clone()
		bamboo.Parent = model
		if i == 1 then
			model.PrimaryPart = bamboo.TallPart
		end
		bamboo:PivotTo(CFrame.new(Vector3.new(bamboo.PrimaryPart.Position.X , i * ySize , bamboo.PrimaryPart.Position.Z)))
	end

	local grass = chunkPart.Grass
	SF.finalizePosition(model, grass, 0.09, 0)
	SF.runGoldenChance(model, plotFolder)
	SF.runRainbowChance(model, plotFolder)
end



function module.makeDefaultTree(chunkPart, biome, plotFolder)
	local model = SF.makePartModel(chunkPart, "Default_Tree")
	
	local trunk, thicknessSize = SF.makeTrunk(model, biome)

	for i = 1, math.random(7,8) do
		local leaf
		
		if biomesFolder[biome]:FindFirstChild("TreeLeaf2") then
			leaf = math.random(1,2) == 1 and biomesFolder[biome].TreeLeaf:Clone() or biomesFolder[biome].TreeLeaf2:Clone()
		else
			leaf = biomesFolder[biome].TreeLeaf:Clone()
		end

		leaf.Parent = model
		
		leaf.Position = Vector3.new(trunk.Position.X + RF.getRandomFloat(-4, 4), trunk.Position.Y + trunk.Size.Y/2 + RF.getRandomFloat(-1.5, 1.5), trunk.Position.Z + RF.getRandomFloat(-4, 4))
		leaf.Size = Vector3.new(RF.getRandomFloat(6, 10), RF.getRandomFloat(5, 7.5), RF.getRandomFloat(6, 10))
		SF.colorOffsetPart(leaf, -5, 45)
	end

	local upperStemTable = SF.make4SideBranch(trunk, biomesFolder[biome].Wood, thicknessSize, "Default_Tree", 1.5, 4, 0)

	for i, v in upperStemTable do
		local upperPos = v.Position.Y + v.Size.Y/2
		local leaf
	
		if biomesFolder[biome]:FindFirstChild("TreeLeaf2") then
			leaf = math.random(1,2) == 1 and biomesFolder[biome].TreeLeaf:Clone() or biomesFolder[biome].TreeLeaf2:Clone()
		else
			leaf = biomesFolder[biome].TreeLeaf:Clone()
		end
		
		leaf.Parent = model
		leaf.Position = Vector3.new(v.Position.X, upperPos, v.Position.Z)
		
		for i = 1, math.random(2,3) do
			local positionOffset = Vector3.new(RF.getRandomFloat(-1,1), RF.getRandomFloat(-1,1), RF.getRandomFloat(-1,1))
			local sizeOffset = Vector3.new(RF.getRandomFloat(-0.3,0.3), RF.getRandomFloat(-0.3, 0.3), RF.getRandomFloat(-0.3,0.3))
			SF.studMaker(leaf, model, positionOffset, sizeOffset, -5, 45)
		end
	end
	

	local grass = chunkPart.Grass
	model.PrimaryPart = trunk
	SF.finalizePosition(model, grass, 0.08, (trunk.Size.Y/2))
	SF.runGoldenChance(model, plotFolder)
	SF.runRainbowChance(model, plotFolder)
	---SF.makeHoly(model, plotFolder)
end

	local grass = chunkPart.Grass

	SF.finalizePosition(model, grass, 0.06, (trunk.Size.Y/2))
	SF.runGoldenChance(model, plotFolder)
	SF.runRainbowChance(model, plotFolder)
end



function module.makeSavanahTree(chunkPart, biome, plotFolder)
	local model = SF.makePartModel(chunkPart, "Slanted_Tree")
	local referenceLeaf = biomesFolder[biome].TreeLeaf
	local referenceWood = biomesFolder[biome].Wood
	local thicknessSize = RF.getRandomFloat(3, 4)

	local function createWoodSegment(size)
		local segment = referenceWood:Clone()
		segment.Size = size
		segment.Parent = model
		return segment
	end

	local bottomWoodLength = RF.getRandomFloat(9, 13)
	local middleWoodLength = bottomWoodLength * 0.7 + thicknessSize
	local topWoodLength = bottomWoodLength * RF.getRandomFloat(0.5,0.9)
	
	local bottomWood = createWoodSegment(Vector3.new(thicknessSize, bottomWoodLength, thicknessSize))
	local middleWood = createWoodSegment(Vector3.new(thicknessSize, thicknessSize, middleWoodLength))
	local topWood = createWoodSegment(Vector3.new(thicknessSize, topWoodLength, thicknessSize))
		
	middleWood.Position = Vector3.new(bottomWood.Position.X + middleWoodLength/2 - thicknessSize/2, bottomWood.Position.Y + bottomWoodLength/2 + thicknessSize/2, bottomWood.Position.Z)
	topWood.Position = Vector3.new(middleWood.Position.X + middleWoodLength/2 - thicknessSize/2, middleWood.Position.Y + topWoodLength/2 + thicknessSize/2, middleWood.Position.Z)

	local topOfTree = topWood.Position + Vector3.new(0, topWood.Size.Y/2, 0	)	
	SF.makeSavanahLeaf(model, biome, topOfTree)
	
	local grass = chunkPart.Grass
	model.PrimaryPart = bottomWood
	SF.finalizePosition(model, grass, 0.02, bottomWood.Size.Y/2, math.random(1,4))
	SF.runGoldenChance(model, plotFolder)
	SF.runRainbowChance(model, plotFolder)
end





function module.makePineConeTree(chunkPart, biome, plotFolder)
	local model = SF.makePartModel(chunkPart, "Pine_Cone_Tree")

	local trunk = biomesFolder[biome].Wood:Clone()
	trunk.Size = Vector3.new(2.3, 15, 2.3)
	trunk.Parent = model
	local leafLayers = math.random(8,12)
	local startingSize = math.random(3,12)
	for i = 1 , leafLayers do
		local leaf = biomesFolder[biome].TreeLeaf:Clone()
		leaf.Parent = model
		local size = startingSize - (i - 1) * 1
		leaf.Position = Vector3.new(trunk.Position.X,trunk.Position.Y + i * 1.3, trunk.Position.Z) 
		leaf.Size = Vector3.new(size , 2 , size)

	end

	local grass = chunkPart.Grass
	model.PrimaryPart = trunk

	SF.finalizePosition(model, grass, 0.02, (trunk.Size.Y/2))
	SF.runGoldenChance(model, plotFolder)
	SF.runRainbowChance(model, plotFolder)
end

end




function module.makeBerryBush(chunkPart, biome, plotFolder)
	local berryChanceList = moduleData.BerryBiomeChance[biome]
	local summedNumber = 0

	for i,v in berryChanceList do
		summedNumber += v
	end


	local berryType = moduleData.BerryNumber[SF.findListPos(math.random(1, summedNumber), 1, berryChanceList)]
	local model = SF.makePartModel(chunkPart, berryType.."_Berry_Bush")


	local bush = SF.studMaker(
		biomesFolder[biome].Bush,
		model,
		Vector3.new(0,0,0),
		Vector3.new(RF.getRandomFloat(0, 2), RF.getRandomFloat(-.5, 0), RF.getRandomFloat(0, 2)),
		-10,
		40
	)
	
	SF.makeBlockSpecs(bush, 6, 8, model, "Berries", berryType)
	

	local grass = chunkPart.Grass
	model.PrimaryPart = bush

	SF.finalizePosition(model, grass, 0.04, (bush.Size.Y/2))
	SF.runGoldenChance(model, plotFolder)
	SF.runRainbowChance(model, plotFolder)
end
