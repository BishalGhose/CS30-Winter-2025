function colorOffsetPart(part, bottomOffset, topOffset)
	local colorOffset = math.random(bottomOffset, topOffset)
	part.Color = Color3.fromRGB(math.clamp(part.Color.R*255 + colorOffset, 0 , 255), math.clamp(part.Color.G*255 + colorOffset, 0 , 255), math.clamp(part.Color.B*255 + colorOffset, 0 , 255))
end

function studMaker(part, model, positionVector, sizeVector, bottomColorOffset, topColorOffset)
	local stud = part:Clone()
	stud.Parent = model
	stud.Position = stud.Position + positionVector
	stud.Size = stud.Size + sizeVector
	colorOffsetPart(stud, bottomColorOffset, topColorOffset)
end

local module = {}

function module.makeMiniRock(chunkPart, biome, shape, heightDifference, scaleFactor)
	local model = Instance.new("Model")
	model.Parent = chunkPart
	model.Name = shape.."Rock"
	
	local grass = chunkPart.Grass


	if shape == "normal" then ---------------------------------------------------------------------
		
		for i = 1, math.random(2,3) do
			studMaker(
				BiomesFolder[biome].Stud,
				model,
				Vector3.new(math.random(-700,700)/100, math.random(-500,500)/1000 + heightDifference, math.random(-700,700)/100),
				Vector3.new(math.random(10,70)/10, math.random(10,30)/10, math.random(10,70)/10),
				-20,
				20
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y + heightDifference, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
		model:ScaleTo((math.random(110,140)/100)^2)

	elseif shape == "spikey" then ---------------------------------------------------------------------
		for i = 1, math.random(3,4) do
			studMaker(
				BiomesFolder[biome].Stud,
				model,
				Vector3.new(math.random(-300,300)/100, math.random(-100,800)/100, math.random(-300,300)/100),
				Vector3.new(math.random(-15,15)/10, math.random(10,30), math.random(-15,15)/10),
				-35,
				15
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y + heightDifference, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
		model:ScaleTo(((math.random(88,160)/100)^2)*scaleFactor)

	elseif shape == "flat" then
		for i = 1, math.random(1,2) do
			studMaker(
				BiomesFolder[biome].Stud,
				model,
				Vector3.new(math.random(-300,300)/100, math.random(-100,800)/100, math.random(-300,300)/100),
				Vector3.new(math.random(130,280)/10, math.random(13,26)/10, math.random(130,280)/10),
				-35,
				15
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y - heightDifference, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
		model:ScaleTo(((math.random(100,120)/100)^2)*scaleFactor)
	end
end

function module.makeCactus(chunkPart, biome)
	local function updateCactusBorders(cactusPart, offset)
		cactusPart.bordersidey.Size = Vector3.new(cactusPart.Size.X - offset, cactusPart.Size.Y + offset, cactusPart.Size.Z - offset)
		cactusPart.bordersidex.Size = Vector3.new(cactusPart.Size.X + offset, cactusPart.Size.Y - offset, cactusPart.Size.Z - offset)
		cactusPart.bordersidez.Size = Vector3.new(cactusPart.Size.X - offset, cactusPart.Size.Y - offset, cactusPart.Size.Z + offset)
	end

	local model = Instance.new("Model")
	model.Parent = chunkPart
	model.Name = "Cactus"

	local mainCactusPart = BiomesFolder["DesertGrass"].CactusPart:Clone()
	mainCactusPart.Parent = model
	mainCactusPart.Name = "MainCactusPart"

	local cactusThickness = math.random(20,30)/10
	local cactusYSize = (math.random(190,280)/100)^2 + cactusThickness*2

	mainCactusPart.Size = Vector3.new(cactusThickness, cactusYSize, cactusThickness)
	updateCactusBorders(mainCactusPart, .5)
	
	
	for i = 1,4 do
		if math.random(1,2) == 1 then
			local startingYPos = mainCactusPart.Position.Y + math.random(0, mainCactusPart.Size.Y/4)
		
			local stem = BiomesFolder["DesertGrass"].CactusPart:Clone()
			local stemModel = Instance.new("Model")
			stemModel.Parent = mainCactusPart
			stem.Parent = stemModel
			stemModel.PrimaryPart = stem
		
			local upperStem = BiomesFolder["DesertGrass"].CactusPart:Clone()
			local upperStemModel = Instance.new("Model")
			upperStemModel.Parent = mainCactusPart
			upperStem.Parent = upperStemModel
			upperStemModel.PrimaryPart = upperStem
			
			local stemThickness = cactusThickness - 2*cactusThickness/5
			local stemLength = cactusThickness * math.random(14,17)/10
			upperStem.Size = Vector3.new(stemThickness, stemLength * math.random(50,80)/100, stemThickness)





			if i == 1 then
				stem.Size = Vector3.new(stemThickness, stemThickness, stemLength)
				local upperStemOffset = stem.Size.Z/2 - upperStem.Size.Z/2
				
				stemModel:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos, stem.Position.Z + stemLength/2)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos + upperStem.Size.Y/2, stem.Position.Z + upperStemOffset)))
			elseif i == 2 then
				stem.Size = Vector3.new(stemThickness, stemThickness, stemLength)
				local upperStemOffset = stem.Size.Z/2 - upperStem.Size.Z/2
				
				stemModel:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos, stem.Position.Z - stemLength/2)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos + upperStem.Size.Y/2, stem.Position.Z - upperStemOffset)))
			elseif i == 3 then
				stem.Size = Vector3.new(stemLength, stemThickness, stemThickness)
				local upperStemOffset = stem.Size.X/2 - upperStem.Size.X/2

				stemModel:PivotTo(CFrame.new(Vector3.new(stem.Position.X + stemLength/2, startingYPos, stem.Position.Z)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X + upperStemOffset, startingYPos + upperStem.Size.Y/2, stem.Position.Z)))
			else
				stem.Size = Vector3.new(stemLength, stemThickness, stemThickness)	
				local upperStemOffset = stem.Size.X/2 - upperStem.Size.X/2

				stemModel:PivotTo(CFrame.new(Vector3.new(stem.Position.X - stemLength/2, startingYPos, stem.Position.Z)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X - upperStemOffset, startingYPos + upperStem.Size.Y/2, stem.Position.Z)))
			end

			updateCactusBorders(stem, .3)
			updateCactusBorders(upperStem, .3)

		end
	end
	
	
	local grass = chunkPart.Grass
	model.PrimaryPart = mainCactusPart
	model:ScaleTo(((math.random(110,170)/100)^2))
	model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y + mainCactusPart.Size.Y/2, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
end

function module.makeBamboo(chunkPart, biome)
	local model = Instance.new("Model")
	model.Parent = chunkPart
	model.Name = "Bamboo"
	

	local ySize = ReplicatedStorage.Biomes[biome].Bamboo.TallPart.Size.Y
	local numberOfParts = math.random(1,3)
	
	for i = 1 , numberOfParts do
		local bamboo = ReplicatedStorage.Biomes[biome].Bamboo:Clone()
		bamboo.Parent = model
		
		bamboo:PivotTo(CFrame.new(Vector3.new(bamboo.PrimaryPart.Position.X , i * ySize , bamboo.PrimaryPart.Position.Z)))
	end

	local grass = chunkPart.Grass
	
	local scale = (math.random(70,170)/100)^2
	model:ScaleTo(scale)
	model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y + ySize * scale * numberOfParts/2, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
end


function module.makeDefaultTree(chunkPart, biome)
	local model = Instance.new("Model")
	model.Parent = chunkPart
	model.Name = "Tree"
	
	local trunk = BiomesFolder[biome].Wood:Clone()
	local thicknessSize = math.random(220,300)/100
	local verticalSize = math.random(50,70)/5 + thicknessSize*2 
	trunk.Size = Vector3.new(thicknessSize, verticalSize, thicknessSize)
	trunk.Parent = model

	for i = 1, math.random(7,8) do
		local leaf
		
		if BiomesFolder[biome]:FindFirstChild("TreeLeaf2") then
			if math.random(1,2) == 1 then
				leaf = BiomesFolder[biome].TreeLeaf:Clone()
			else
				leaf = BiomesFolder[biome].TreeLeaf2:Clone()
			end
		else
			leaf = BiomesFolder[biome].TreeLeaf:Clone()
		end

		leaf.Parent = model
		
		leaf.Position = Vector3.new(trunk.Position.X + math.random(-400, 400)/100, trunk.Position.Y + trunk.Size.Y/2 + math.random(-150, 150)/100, trunk.Position.Z + math.random(-400, 400)/100)
		leaf.Size = Vector3.new(math.random(300,500)/50, math.random(250,380)/50, math.random(300,500)/50)
		colorOffsetPart(leaf, -5, 45)
	end

	
	
	local grass = chunkPart.Grass
	model.PrimaryPart = trunk
	model:ScaleTo(((math.random(100,150)/100)^2))
	model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y + trunk.Size.Y/2, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
end
