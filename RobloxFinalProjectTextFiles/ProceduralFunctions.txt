function module.makeMiniRock(chunkPart, biome, shape, heightDifference, scaleFactor)
	local model = CF.makePartModel(chunkPart, shape.."_Rock")
	
	local grass = chunkPart.Grass


	if shape == "normal" then ---------------------------------------------------------------------
		
		for i = 1, math.random(2,3) do
			CF.studMaker(
				biomesFolder[biome].Stud,
				model,
				Vector3.new(CF.getRandomFloat(-7, 7), CF.getRandomFloat(-.5, .5) + heightDifference, CF.getRandomFloat(-7, 7)),
				Vector3.new(CF.getRandomFloat(1, 7), CF.getRandomFloat(1, 3), CF.getRandomFloat(1, 7)),
				-20,
				20
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y + heightDifference, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
		model:ScaleTo(CF.getExponentialScale(0.06))

	elseif shape == "spikey" then ---------------------------------------------------------------------
		for i = 1, math.random(3,4) do
			CF.studMaker(
				biomesFolder[biome].Stud,
				model,
				Vector3.new(CF.getRandomFloat(-4, 4), CF.getRandomFloat(-1, 8), CF.getRandomFloat(-4, 4)),
				Vector3.new(CF.getRandomFloat(-3, 3), CF.getRandomFloat(15, 35), CF.getRandomFloat(-3, 3)),
				-35,
				15
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y + heightDifference, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
		model:ScaleTo((CF.getExponentialScale(0.07))*scaleFactor)

	elseif shape == "flat" then ---------------------------------------------------------------------
		for i = 1, math.random(1,2) do
			CF.studMaker(
				biomesFolder[biome].Stud,
				model,
				Vector3.new(CF.getRandomFloat(-3, 3), CF.getRandomFloat(-2, 2), CF.getRandomFloat(-3, 3)),
				Vector3.new(CF.getRandomFloat(13, 23), CF.getRandomFloat(1.3, 2.3), CF.getRandomFloat(13, 23)),
				-35,
				15
			)
		end


		model.PrimaryPart = model:GetChildren()[1]
		model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + math.random(-grass.Size.X/3, grass.Size.X/3), grass.Position.Y - heightDifference, grass.Position.Z + math.random(-grass.Size.Z/3, grass.Size.Z/3))))
		model:ScaleTo((CF.getExponentialScale(0.06))*scaleFactor)
	end
	
	CF.runGoldenChance(model)
	CF.runRainbowChance(model)
end

function module.makeCactus(chunkPart, biome)
	function CF.updateCactusBorders(cactusPart, offset)
		cactusPart.bordersidey.Size = Vector3.new(cactusPart.Size.X - offset, cactusPart.Size.Y + offset, cactusPart.Size.Z - offset)
		cactusPart.bordersidex.Size = Vector3.new(cactusPart.Size.X + offset, cactusPart.Size.Y - offset, cactusPart.Size.Z - offset)
		cactusPart.bordersidez.Size = Vector3.new(cactusPart.Size.X - offset, cactusPart.Size.Y - offset, cactusPart.Size.Z + offset)
	end

	local model = CF.makePartModel(chunkPart, "Cactus")

	local mainCactusPart = biomesFolder["DesertGrass"].CactusPart:Clone()
	mainCactusPart.Parent = model
	mainCactusPart.Name = "MainCactusPart"

	local cactusThickness = CF.getRandomFloat(2, 3)
	local cactusYSize = (CF.getRandomFloat(1.9, 2.8))^2 + cactusThickness*2

	mainCactusPart.Size = Vector3.new(cactusThickness, cactusYSize, cactusThickness)
	CF.updateCactusBorders(mainCactusPart, 0.5)
	
	
	CF.make4SideBranch(mainCactusPart,  biomesFolder.DesertGrass["CactusPart"], cactusThickness, "Cactus", 1, 2, -mainCactusPart.Size.Y/4)
	
	
	local grass = chunkPart.Grass
	model.PrimaryPart = mainCactusPart
	CF.finalizePosition(model, grass, 0.06, mainCactusPart.Size.Y/2)
	CF.runGoldenChance(model)
	CF.runRainbowChance(model)
end

function module.makeBamboo(chunkPart, biome)
	local model = CF.makePartModel(chunkPart, "Bamboo")
	

	local ySize = biomesFolder[biome].Bamboo.TallPart.Size.Y
	local numberOfParts = math.random(1,3)
	
	for i = 1 , numberOfParts do
		local bamboo = biomesFolder[biome].Bamboo:Clone()
		bamboo.Parent = model
		if i == 1 then
			model.PrimaryPart = bamboo.TallPart
		end
		bamboo:PivotTo(CFrame.new(Vector3.new(bamboo.PrimaryPart.Position.X , i * ySize , bamboo.PrimaryPart.Position.Z)))
	end

	local grass = chunkPart.Grass
	CF.finalizePosition(model, grass, 0.09, 0)
	CF.runGoldenChance(model)
	CF.runRainbowChance(model)
end



function module.makeDefaultTree(chunkPart, biome)
	local model = CF.makePartModel(chunkPart, "Default_Tree")
	
	local trunk, thicknessSize = CF.makeTrunk(model, biome)

	for i = 1, math.random(7,8) do
		local leaf
		
		if biomesFolder[biome]:FindFirstChild("TreeLeaf2") then
			leaf = math.random(1,2) == 1 and biomesFolder[biome].TreeLeaf:Clone() or biomesFolder[biome].TreeLeaf2:Clone()
		else
			leaf = biomesFolder[biome].TreeLeaf:Clone()
		end

		leaf.Parent = model
		
		leaf.Position = Vector3.new(trunk.Position.X + CF.getRandomFloat(-4, 4), trunk.Position.Y + trunk.Size.Y/2 + CF.getRandomFloat(-1.5, 1.5), trunk.Position.Z + CF.getRandomFloat(-4, 4))
		leaf.Size = Vector3.new(CF.getRandomFloat(6, 10), CF.getRandomFloat(5, 7.5), CF.getRandomFloat(6, 10))
		CF.colorOffsetPart(leaf, -5, 45)
	end

	local upperStemTable = CF.make4SideBranch(trunk, biomesFolder[biome].Wood, thicknessSize, "Default_Tree", 1.5, 4, 0)

	for i, v in upperStemTable do
		local upperPos = v.Position.Y + v.Size.Y/2
		local leaf
	
		if biomesFolder[biome]:FindFirstChild("TreeLeaf2") then
			leaf = math.random(1,2) == 1 and biomesFolder[biome].TreeLeaf:Clone() or biomesFolder[biome].TreeLeaf2:Clone()
		else
			leaf = biomesFolder[biome].TreeLeaf:Clone()
		end
		
		leaf.Parent = model
		leaf.Position = Vector3.new(v.Position.X, upperPos, v.Position.Z)
		
		for i = 1, math.random(2,3) do
			local positionOffset = Vector3.new(CF.getRandomFloat(-1,1), CF.getRandomFloat(-1,1), CF.getRandomFloat(-1,1))
			local sizeOffset = Vector3.new(CF.getRandomFloat(-0.3,0.3), CF.getRandomFloat(-0.3, 0.3), CF.getRandomFloat(-0.3,0.3))
			CF.studMaker(leaf, model, positionOffset, sizeOffset, -5, 45)
		end
	end
	

	local grass = chunkPart.Grass
	model.PrimaryPart = trunk
	CF.finalizePosition(model, grass, 0.08, (trunk.Size.Y/2))
	CF.runGoldenChance(model)
	CF.runRainbowChance(model)
end

function module.makeSavanahTree(chunkPart, biome)
	local model = CF.makePartModel(chunkPart, "Slanted_Tree")
	local referenceLeaf = biomesFolder[biome].TreeLeaf
	local referenceWood = biomesFolder[biome].Wood
	local thicknessSize = CF.getRandomFloat(3, 4)

	local function createWoodSegment(size)
		local segment = referenceWood:Clone()
		segment.Size = size
		segment.Parent = model
		return segment
	end

	local bottomWoodLength = CF.getRandomFloat(9, 13)
	local middleWoodLength = bottomWoodLength * 0.7 + thicknessSize
	local topWoodLength = bottomWoodLength * CF.getRandomFloat(0.5,0.9)
	
	local bottomWood = createWoodSegment(Vector3.new(thicknessSize, bottomWoodLength, thicknessSize))
	local middleWood = createWoodSegment(Vector3.new(thicknessSize, thicknessSize, middleWoodLength))
	local topWood = createWoodSegment(Vector3.new(thicknessSize, topWoodLength, thicknessSize))
		
	middleWood.Position = Vector3.new(bottomWood.Position.X + middleWoodLength/2 - thicknessSize/2, bottomWood.Position.Y + bottomWoodLength/2 + thicknessSize/2, bottomWood.Position.Z)
	topWood.Position = Vector3.new(middleWood.Position.X + middleWoodLength/2 - thicknessSize/2, middleWood.Position.Y + topWoodLength/2 + thicknessSize/2, middleWood.Position.Z)

	local topOfTree = topWood.Position + Vector3.new(0, topWood.Size.Y/2, 0	)	
	CF.makeSavanahLeaf(model, biome, topOfTree)
	
	local grass = chunkPart.Grass
	model.PrimaryPart = bottomWood
	CF.finalizePosition(model, grass, 0.02, (bottomWood.Size.Y/2), math.random(1,4))
	CF.runGoldenChance(model)
	CF.runRainbowChance(model)
end





function module.makePineConeTree(chunkPart, biome)
	local model = CF.makePartModel(chunkPart, "Pine_Cone_Tree")

	local trunk = biomesFolder[biome].Wood:Clone()
	trunk.Parent = model
	local leafLayers = math.random(8,12)
	local startingSize = math.random(9,12)
	for i = 1 , leafLayers do
		local leaf = biomesFolder[biome].TreeLeaf:Clone()
		leaf.Parent = model
		local size = startingSize - (i - 1) * 1
		leaf.Position = Vector3.new(trunk.Position.X, trunk.Position.Y + i * 2, trunk.Position.Z) 
		leaf.Size = Vector3.new(size , 2 , size)

	end

	local grass = chunkPart.Grass
	model.PrimaryPart = trunk

	CF.finalizePosition(model, grass, 0.02, (trunk.Size.Y/2))
	CF.runGoldenChance(model)
	CF.runRainbowChance(model)
end

function module.makeBerryBush(chunkPart, biome)
	local berryChanceList = ModuleData.BerryBiomeChance[biome]
	local summedNumber = 0

	for i,v in berryChanceList do
		summedNumber += v
	end


	local berryType = ModuleData.BerryNumber[CF.findListPos(math.random(1, summedNumber), 1, berryChanceList)]
	local model = CF.makePartModel(chunkPart, berryType.."_Berry_Bush")


	local bush = CF.studMaker(
		biomesFolder[biome].Bush,
		model,
		Vector3.new(0,0,0),
		Vector3.new(CF.getRandomFloat(0, 2), CF.getRandomFloat(-.5, 0), CF.getRandomFloat(0, 2)),
		-10,
		40
	)
	
	CF.makeBlockSpecs(bush, 6, 8, model, "Berries", berryType)
	

	local grass = chunkPart.Grass
	model.PrimaryPart = bush

	CF.finalizePosition(model, grass, 0.04, (bush.Size.Y/2))
	CF.runGoldenChance(model)
	CF.runRainbowChance(model)
end
