function module.getExponentialScale(rarity)
	local u = math.random()
	return 1 + 10 * (1 - u^rarity)
end

function module.colorOffsetPart(part, bottomOffset, topOffset)
	local colorOffset = math.random(bottomOffset, topOffset)
	part.Color = Color3.fromRGB(math.clamp(part.Color.R*255 + colorOffset, 0 , 255), math.clamp(part.Color.G*255 + colorOffset, 0 , 255), math.clamp(part.Color.B*255 + colorOffset, 0 , 255))
end


function module.studMaker(part, model, positionVector, sizeVector, bottomColorOffset, topColorOffset)
	local stud = part:Clone()
	stud.Parent = model
	stud.Position = stud.Position + positionVector
	stud.Size = stud.Size + sizeVector
	module.colorOffsetPart(stud, bottomColorOffset, topColorOffset)
	return stud
end

function module.findListPos(number, pos, list)
	if number <= list[pos] then 
		return 1
	end

	return 1 + module.findListPos(number - list[pos], pos + 1, list)
end

function module.makeTrunk(model, biome)
	local trunk = biomesFolder[biome].Wood:Clone()
	local thicknessSize = RF.getRandomFloat(2.2, 3)
	local verticalSize = RF.getRandomFloat(10, 14) + thicknessSize*2 
	trunk.Size = Vector3.new(thicknessSize, verticalSize, thicknessSize)
	trunk.Parent = model
	return trunk, thicknessSize
end


function module.makeBlockSpecs(part, lowerAmount, upperAmount, model, speckType, speckVariation)
	local xSize = part.Size.X
	local ySize = part.Size.Y
	local zSize = part.Size.Z


	for side = 0, 1 do 
		for axis = 1, 3 do
			for amount = 1, math.random(1,2) do
				local posX, posY, posZ

				if axis == 1 then
					posX = side == 0 and -9*xSize/20 or 9*xSize/20
					posY = -ySize/3 + math.random() * 2*ySize/3
					posZ = -zSize/3 + math.random() * 2*zSize/3
				elseif axis == 2 then
					posX = -xSize/3 + math.random() * 2*xSize/3
					posY = 9*ySize/20
					posZ = -zSize/3 + math.random() * 2*zSize/3
				elseif axis == 3 then
					posX = -xSize/3 + math.random() * 2*xSize/3
					posY = -ySize/3 + math.random() * 2*ySize/3
					posZ = side == 0 and -9*zSize/20 or 9*zSize/20
				end

				local speck
				if speckType == "Ores" then
					speck = ReplicatedStorage[speckType][speckVariation.."_Ore"]:Clone()
					speck.Size *= RF.getRandomFloat(1, 1.7)
				elseif speckType == "Berries" then
					speck = ReplicatedStorage[speckType][speckVariation.."_Berry"]:Clone()
				end

				speck.Parent = model
				speck.Position = Vector3.new(part.Position.X + posX, part.Position.Y + posY, part.Position.Z + posZ)
			end
		end
	end
end



function module.make4SideBranch(trunk, stemReferencePart, trunkThickness, treeType, stemLengthMultiplier, branchChance, bottomYLimit)
	local upperStemTable = {}
	for i = 1,4 do
		if math.random(1, branchChance) == 1 then
			local startingYPos = trunk.Position.Y + RF.getRandomFloat(bottomYLimit, trunk.Size.Y/4)

			local stem = stemReferencePart:Clone()
			stem.Parent = trunk

			local upperStem = stemReferencePart:Clone()
			upperStem.Parent = trunk

			local stemThickness = trunkThickness - 2*trunkThickness/5
			local stemLength = trunkThickness * RF.getRandomFloat(1.4, 1.7) * stemLengthMultiplier
			upperStem.Size = Vector3.new(stemThickness, stemLength * RF.getRandomFloat(0.5, 0.8), stemThickness)



			if i == 1 then
				stem.Size = Vector3.new(stemThickness, stemThickness, stemLength)
				local upperStemOffset = stem.Size.Z/2 - upperStem.Size.Z/2

				stem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos, stem.Position.Z + stemLength/2)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos + upperStem.Size.Y/2, stem.Position.Z + upperStemOffset)))
			elseif i == 2 then
				stem.Size = Vector3.new(stemThickness, stemThickness, stemLength)
				local upperStemOffset = stem.Size.Z/2 - upperStem.Size.Z/2

				stem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos, stem.Position.Z - stemLength/2)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos + upperStem.Size.Y/2, stem.Position.Z - upperStemOffset)))
			elseif i == 3 then
				stem.Size = Vector3.new(stemLength, stemThickness, stemThickness)
				local upperStemOffset = stem.Size.X/2 - upperStem.Size.X/2

				stem:PivotTo(CFrame.new(Vector3.new(stem.Position.X + stemLength/2, startingYPos, stem.Position.Z)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X + upperStemOffset, startingYPos + upperStem.Size.Y/2, stem.Position.Z)))
			else
				stem.Size = Vector3.new(stemLength, stemThickness, stemThickness)	
				local upperStemOffset = stem.Size.X/2 - upperStem.Size.X/2

				stem:PivotTo(CFrame.new(Vector3.new(stem.Position.X - stemLength/2, startingYPos, stem.Position.Z)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X - upperStemOffset, startingYPos + upperStem.Size.Y/2, stem.Position.Z)))
			end

			table.insert(upperStemTable, upperStem)

			if treeType == "Cactus" then
				module.updateCactusBorders(stem, .3)
				module.updateCactusBorders(upperStem, .3)
			end
		end
	end
	return upperStemTable
end

function module.runGoldenChance(model, plotFolder)
	if math.random(1,64) == 1 then
		CollectionService:AddTag(model, "Golden")
		model.Parent = plotFolder.GoldenParts
		for i,v in model:GetDescendants() do
			if v:IsA("Part") then
				v.Color = Color3.fromRGB(255, 215, 0)
				module.colorOffsetPart(v, -25, 25)
			end
			local goldenParticle = particleFolder.GoldenParticles:Clone()
			goldenParticle.Parent = v
			if v == model.PrimaryPart then
				goldenParticle.Rate = 18
			end
		end
	end
end

end

function module.finalizePosition(model, grass, scaleFactor, yOffset, rotationMultiplier)
	local scale = module.getExponentialScale(scaleFactor)
	model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + RF.getRandomFloat(-grass.Size.X*0.4, grass.Size.X*0.4), grass.Position.Y + yOffset*scale , grass.Position.Z + RF.getRandomFloat(-grass.Size.Z*0.4, grass.Size.Z*0.4))))
	model:ScaleTo(scale)
	model:SetAttribute("Scale", scale)

	for i, v in model.PrimaryPart:GetTouchingParts() do
		if v ~= model.Parent and v.Parent.Name == "MainParts" then
			local offset = math.sqrt(v:GetAttribute("yValue") - model.Parent:GetAttribute("yValue"))
			local pivot = model:GetPivot()
			local newPivot

			if rotationMultiplier then
				newPivot = pivot * CFrame.new(0, offset, 0) * CFrame.Angles(0, math.rad(90 * rotationMultiplier), 0)
			else
				newPivot = pivot * CFrame.new(0, offset, 0)
			end
			model:PivotTo(newPivot)
		end
	end

end

function module.makeMap(plotNumber, tableOfCoords, player)
	local userId = player.UserId
	local heightScale = RF.getRandomFloat(.022, .026)
	importantValues.HeightScale.Value = heightScale
	local plotFolder = plot["Plot"..plotNumber]
	

	for biomeName, chanceData in moduleData.BiomeChanceData do
		local biomeSeed
		local savedBiomeSeed = playerData["BiomeSeeds"][player.UserId][biomeName]
		if savedBiomeSeed then
			biomeSeed = savedBiomeSeed
		else
			biomeSeed = math.random(1,100000000)
			playerData["BiomeSeeds"][player.UserId][biomeName] = biomeSeed
		end
	end


	local heightSeed
	local savedHeightSeed = playerData["HeightSeed"][player.UserId][1]
	if savedHeightSeed then
		heightSeed = savedHeightSeed
	else
		heightSeed = math.random(1,100000000)
		while module.checkHeightSeed(heightScale, heightSeed, plotNumber, {1,1}) == false do
			heightSeed = math.random(1,100000000)
		end
		playerData["HeightSeed"][player.UserId][1] = heightSeed
	end

	for i,v in tableOfCoords do
		module.AddChunk(v[1], v[2], plotNumber, heightSeed, heightScale, false, player)
	end


	for i,v in RF.getBuyCoordinates(playerData["OwnedChunks"][userId]) do
		if chunkSizes[userId][v[1]..","..v[2]] == nil then
			module.setChunkSize(v[1], v[2], plotNumber, heightSeed, heightScale, userId)
		end
	end
	updateClientData:FireClient(player, "ChunkSizes", chunkSizes[userId])


	local ownedChunks = playerData["OwnedChunks"][player.UserId]
	updateClientData:FireClient(player, "OwnedChunks", ownedChunks)
end



function module.checkHeightSeed(scale, seed, plotNumber, coords)
	local x, z = RF.coordinateScale(increment, plotNumber, coords)

	local noiseLocation = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)
	if noiseLocation < -0.5 or noiseLocation > 0.5 then
		return false
	else
		return true
	end
end


function module.AddChunk(coord1, coord2, plotNumber, heightSeed, heightScale, animation, player)
	local userId = player.UserId
	local part = chunkBlock:Clone()
	local plotFolder = plot["Plot"..plotNumber]
	local fullCoord = coord1..","..coord2..","..plotNumber
	part.Name = fullCoord

	
	module.makeHeightNoise(coord1, coord2, heightScale, heightSeed, plotNumber, part, userId)


	for i,v in playerData.BiomeSeeds[player.UserId] do
		module.makeBiomeNoise(coord1, coord2, biomeScale, i, v, part, plotNumber)
	end

	module.checkEmptyChunk(part)

	if animation then
		part.Parent = workspace.TempStorage
		table.insert(playerData["OwnedChunks"][userId], {coord1, coord2})
		for i,v in RF.getBuyCoordinates(playerData["OwnedChunks"][userId]) do
			if chunkSizes[userId][v[1]..","..v[2]] == nil then
				module.setChunkSize(v[1], v[2], plotNumber, heightSeed, heightScale, userId)
			end
		end
		updateClientData:FireClient(player, "OwnedChunks", playerData["OwnedChunks"][userId])
		updateClientData:FireClient(player, "ChunkSizes", chunkSizes[userId])
		destroySkeleton:FireClient(player, fullCoord)
		

		local top = part:FindFirstChild("Grass") or part:FindFirstChild("WaterBlock")

		local tweenParts = {top, part}
		local tweenInfo = TweenInfo.new(2, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out, 0, false, 0)

		for i,v in ipairs(tweenParts) do
			local originalPosition = v.Position
			v.Position = originalPosition + Vector3.new(0, -20, 0)
			v.Transparency = 1
			local goal = {Transparency = 0, Position = originalPosition}
			local tween = TweenService:Create(v, tweenInfo, goal)

			tween:Play()
			
			if i == 2 then

				local connection
				connection = tween.Completed:Connect(function()
					part.Parent = plotFolder.MainParts
					connection:Disconnect()
					connection = nil
				end)
			end
		end
	end
end



function module.makeHeightNoise(coord1, coord2, scale, seed, plotNumber, part, userId)
	local x, z = RF.coordinateScale(increment, plotNumber, {coord1, coord2})
	local plotFolder = plot["Plot"..plotNumber]
	local noiseValue = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)
	local normalized = noiseValue + 1           
	local y = normalized * 100
	local yMainPartLocation = y/2
	part:SetAttribute("yValue", y)

	local xSize, ySize, zSize, noiseValue = module.setChunkSize(coord1, coord2, plotNumber, seed, scale, userId)

	
	part.Size = Vector3.new(xSize, y, zSize)
	part.Anchored = true
	part.Position = Vector3.new(x, yMainPartLocation, z)
	part.Parent = plotFolder.MainParts

	if noiseValue < -0.4 then
		CollectionService:AddTag(part, "Water")

		part.Material = Enum.Material.Sand
		part.Color = Color3.fromRGB(199, 159, 119)

		module.colorOffsetPart(part, -15, 15)

		local waterBlock = ServerStorage.WaterBlock:Clone()
		waterBlock.Parent = part
		local ySize = 60 - (yMainPartLocation + part.Size.Y/2)
		local yPosition = yMainPartLocation + part.Size.Y/2 + ySize/2
		waterBlock.Size = Vector3.new(part.Size.X, ySize, part.Size.Z)
		waterBlock.Position = Vector3.new(x, yPosition, z)
		module.colorOffsetPart(waterBlock, -15, 15)
	end
end



function module.makeBiomeNoise(coord1, coord2, scale, biomeName, seed, part, plotNumber)
	if CollectionService:HasTag(part, "Water")  then
		return
	end
	
	local x, z = RF.coordinateScale(increment, plotNumber, {coord1, coord2})
		
	local chanceData = moduleData.BiomeChanceData[biomeName]
	scale *= chanceData[2]
	local noiseLocation = math.noise((x + seed) / (increment) * scale, (z + seed) / (increment) * scale)

	if noiseLocation >= chanceData[1] then
		module.replaceBlock(part, ReplicatedStorage.Biomes[biomeName])
	end
end



function module.replaceBlock(part, GrassType)
	for i,v in part:GetChildren() do
		v:Destroy()
	end

	for i,v in moduleData.GrassTypes do
		if part:HasTag(v) then
			CollectionService:RemoveTag(part, v)
		end
	end

	CollectionService:AddTag(part, GrassType.Name)


	local offsetColor1 = math.random(-10,10)/255
	local red = math.clamp(GrassType.Dirt.Color.R + offsetColor1, 0, 1)
	local green = GrassType.Dirt.Color.G
	local blue = math.clamp(GrassType.Dirt.Color.B + offsetColor1, 0, 1)
	part.Color = Color3.new(red, green, blue)
	part.Material = GrassType.Dirt.Material

	local GrassClone = GrassType.Grass:Clone()
	GrassClone.Parent = part
	GrassClone.Size = Vector3.new(part.Size.X, 1, part.Size.Z)
	GrassClone.Position = Vector3.new(part.Position.X, part.Position.Y + part.Size.Y/2 + RF.getRandomFloat(0.45,0.5), part.Position.Z)


	local offsetColor2 = math.random(-15,15)/255

	local red = math.clamp(GrassClone.Color.R + offsetColor2, 0, 1)
	local green = math.clamp(GrassClone.Color.G + offsetColor2, 0, 1)
	local blue = math.clamp(GrassClone.Color.B + offsetColor2, 0, 1)
	GrassClone.Color = Color3.new(red, green, blue)
end



function module.checkEmptyChunk(chunkPart)
	local hasTag = false
	for x,y in moduleData.GrassTypes do
		if chunkPart:HasTag(y) then
			hasTag = true
		end
	end
	if hasTag == false then
		module.replaceBlock(chunkPart, ReplicatedStorage.Biomes.NormalGrass)
		CollectionService:AddTag(chunkPart, "NormalGrass")
	end
end



function module.deleteMap(plotFolder)
	local tableOfPartsFolders = {"MainParts", "GoldenParts", "RainbowParts", "BloodRedParts", "HolyParts", "BuyUIFolder", "SeedDataFolder"}

	for i, v in tableOfPartsFolders do

			if y:IsA("Highlight") == false then
				y:Destroy()
			end
		end
	end

	plotFolder.NumberOfObjects.Value = 0
end

function module.setChunkSize(coord1, coord2, plotNumber, seed, scale, userId)
	local xSize, ySize, zSize, noiseValue
	local savedSizeTable = chunkSizes[userId][coord1..","..coord2]
	if savedSizeTable then
		xSize = savedSizeTable[1]
		ySize = savedSizeTable[2]
		zSize = savedSizeTable[3]
		noiseValue = savedSizeTable[4]

	else
		local x, z = RF.coordinateScale(increment, plotNumber, {coord1, coord2})
		noiseValue = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)         
		ySize = (noiseValue + 1 ) * 100

		xSize = increment + RF.getRandomFloat(0, increment)
		zSize = increment + RF.getRandomFloat(0, increment)
		chunkSizes[userId][coord1..","..coord2] = {xSize, ySize, zSize, noiseValue}
	end
	return	xSize, ySize, zSize, noiseValue
end
