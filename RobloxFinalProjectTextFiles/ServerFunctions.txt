-- A rarity function that returns an umber from 1-10 with a very rare chance for it to be close to 10
function module.getExponentialScale(rarity)
	local u = math.random()
	return 1 + 10 * (1 - u^rarity)
end


-- Takes a part and takes it color and offsets by a random amount in the given range
function module.colorOffsetPart(part, bottomOffset, topOffset)
	local colorOffset = math.random(bottomOffset, topOffset)
	part.Color = Color3.fromRGB(math.clamp(part.Color.R*255 + colorOffset, 0 , 255), math.clamp(part.Color.G*255 + colorOffset, 0 , 255), math.clamp(part.Color.B*255 + colorOffset, 0 , 255))
end


-- Creates a new stud/block. With parameters to offset its color and position from original part
-- Used in trees, bushes, rocks etc
function module.studMaker(part, model, positionVector, sizeVector, bottomColorOffset, topColorOffset)
	local stud = part:Clone()
	stud.Parent = model
	stud.Position = stud.Position + positionVector
	stud.Size = stud.Size + sizeVector
	module.colorOffsetPart(stud, bottomColorOffset, topColorOffset)
	return stud
end


-- Recursively finds the list position of a number where the position would be if
-- You subtract that number to all the numbers in the list till that number isnt bigger than current number(hard to explain)
-- like for example  a list of numbers = [1, 4, 6] and given number 3
-- The position with would be 2 because u minus 3-1 and u have 2 left stud but 2 is not bigger than 4
-- Used for rarity system for bushes/ores
function module.findListPos(number, pos, list)
	if number <= list[pos] then 
		return 1
	end

	return 1 + module.findListPos(number - list[pos], pos + 1, list)
end

-- Makes the trunk from the given biome type
function module.makeTrunk(model, biome)
	local trunk = biomesFolder[biome].Wood:Clone()
	local thicknessSize = RF.getRandomFloat(2.2, 3)

-- Takes a part and draws specks around it
-- Used for berry bushes and ores
function module.makeBlockSpecs(part, model, speckType, speckVariation)
	local xSize = part.Size.X
	local ySize = part.Size.Y
	local zSize = part.Size.Z

	-- Loops through the axises/sides
	for side = 0, 1 do 
		for axis = 1, 3 do
			for amount = 1, math.random(1,2) do
				local posX, posY, posZ
				
				-- Sets the random position variables
				if axis == 1 then
					posX = side == 0 and -9*xSize/20 or 9*xSize/20
					posY = -ySize/3 + math.random() * 2*ySize/3
					posZ = -zSize/3 + math.random() * 2*zSize/3
				elseif axis == 2 then
					posX = -xSize/3 + math.random() * 2*xSize/3
					posY = 9*ySize/20
					posZ = -zSize/3 + math.random() * 2*zSize/3
				elseif axis == 3 then
					posX = -xSize/3 + math.random() * 2*xSize/3
					posY = -ySize/3 + math.random() * 2*ySize/3
					posZ = side == 0 and -9*zSize/20 or 9*zSize/20
				end

				-- Clones the berry/ore type and sets their position and parent
				local speck
				if speckType == "Ores" then
					speck = ReplicatedStorage[speckType][speckVariation.."_Ore"]:Clone()
					speck.Size *= RF.getRandomFloat(1, 1.7)
				elseif speckType == "Berries" then
					speck = ReplicatedStorage[speckType][speckVariation.."_Berry"]:Clone()
				end
				speck.Parent = model
				speck.Position = Vector3.new(part.Position.X + posX, part.Position.Y + posY, part.Position.Z + posZ)
			end
		end
	end
end


-- Draws branches when given a trunk
-- Used for cactus and trees
function module.make4SideBranch(trunk, stemReferencePart, trunkThickness, treeType, stemLengthMultiplier, branchChance, bottomYLimit)
	local upperStemTable = {}

	for i = 1,4 do
		if math.random(1, branchChance) == 1 then -- Sets a chance for branch to occur using branchChance parameter
			local startingYPos = trunk.Position.Y + RF.getRandomFloat(bottomYLimit, trunk.Size.Y/4)

			-- lower and upper stems
			local stem = stemReferencePart:Clone()
			local upperStem = stemReferencePart:Clone()
			stem.Parent = trunk
			upperStem.Parent = trunk
			
			-- Sets some of the properties of the stems
			local stemThickness = trunkThickness - 2*trunkThickness/5
			local stemLength = trunkThickness * RF.getRandomFloat(1.4, 1.7) * stemLengthMultiplier
			upperStem.Size = Vector3.new(stemThickness, stemLength * RF.getRandomFloat(0.5, 0.8), stemThickness)


			-- Basically gets the side and makes a stem extend outwards in that direction
			-- Then gets another stem but its vertical and is on the tip of the first
			-- Stem then offsetted by a little
			if i == 1 then
				stem.Size = Vector3.new(stemThickness, stemThickness, stemLength)
				local upperStemOffset = stem.Size.Z/2 - upperStem.Size.Z/2

				stem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos, stem.Position.Z + stemLength/2)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos + upperStem.Size.Y/2, stem.Position.Z + upperStemOffset)))
			elseif i == 2 then
				stem.Size = Vector3.new(stemThickness, stemThickness, stemLength)
				local upperStemOffset = stem.Size.Z/2 - upperStem.Size.Z/2

				stem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos, stem.Position.Z - stemLength/2)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X, startingYPos + upperStem.Size.Y/2, stem.Position.Z - upperStemOffset)))
			elseif i == 3 then
				stem.Size = Vector3.new(stemLength, stemThickness, stemThickness)
				local upperStemOffset = stem.Size.X/2 - upperStem.Size.X/2

				stem:PivotTo(CFrame.new(Vector3.new(stem.Position.X + stemLength/2, startingYPos, stem.Position.Z)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X + upperStemOffset, startingYPos + upperStem.Size.Y/2, stem.Position.Z)))
			else
				stem.Size = Vector3.new(stemLength, stemThickness, stemThickness)	
				local upperStemOffset = stem.Size.X/2 - upperStem.Size.X/2

				stem:PivotTo(CFrame.new(Vector3.new(stem.Position.X - stemLength/2, startingYPos, stem.Position.Z)))
				upperStem:PivotTo(CFrame.new(Vector3.new(stem.Position.X - upperStemOffset, startingYPos + upperStem.Size.Y/2, stem.Position.Z)))
			end

			table.insert(upperStemTable, upperStem)

			-- If the tree type is cactus then it updates the cactus borders because the cactus
			-- Has like a border thing that needs to be adjusted for the new size
			if treeType == "Cactus" then
				print(module)
				module.updateCactusBorders(stem, .3)
				module.updateCactusBorders(upperStem, .3)
			end
		end
	end
	return upperStemTable
end

-- Takes the borders of the cactus and adjusts it the the new size of the main part of the cactus
function module.updateCactusBorders(cactusPart, offset)
	cactusPart.bordersidey.Size = Vector3.new(cactusPart.Size.X - offset, cactusPart.Size.Y + offset, cactusPart.Size.Z - offset)
	cactusPart.bordersidex.Size = Vector3.new(cactusPart.Size.X + offset, cactusPart.Size.Y - offset, cactusPart.Size.Z - offset)
	cactusPart.bordersidez.Size = Vector3.new(cactusPart.Size.X - offset, cactusPart.Size.Y - offset, cactusPart.Size.Z + offset)
end

-- Function to tween the lighting properties that are tweenable
function module.tweenLighting(weatherEnvironment, tweenInfo)
	for i,v in environments[weatherEnvironment]:GetChildren() do
		local object = Lighting[v.Name]
		for x,y in moduleData.tweenableProperties[v.Name] do
			local goal = {[y] = v[y]}
			local tween = TweenService:Create(object, tweenInfo, goal)
			tween:Play()
		end
	end
end

-- Function to set the lighting properties that are not tweenable
function module.changeLighting(weatherEnvironment)
	for i,v in environments[weatherEnvironment]:GetChildren() do
		local object = Lighting[v.Name]
		for x,y in moduleData.properties[v.Name] do
			object[y] = v[y]
		end
	end
end


-- Function to run the blood moon event
function module.makeBloodMoon(duration)
	local environments = Lighting.Environments
	Lighting.Sky.MoonTextureId = environments.BloodMoonEnvironment.Sky.MoonTextureId
	
	-- This chunk of code makes it so it creates a part around every player where a particle effect is placed on
	-- that part and creates a firefly effect
	local deleteObjects = {}
	for i,v in Players:GetPlayers() do
		local character = v.Character
		if character then
			local fireFliesPart = ReplicatedStorage.FireFliesPart:Clone()
			local weldConstraint = Instance.new("WeldConstraint", character)
			local humanoidRootPart = character.HumanoidRootPart

			-- Saves to delete later when event is over
			table.insert(deleteObjects, fireFliesPart) 
			table.insert(deleteObjects, weldConstraint)
			
			fireFliesPart.Anchored = false
			fireFliesPart.Position = humanoidRootPart.Position
			fireFliesPart.Parent = character
				
			-- Weld the parts together 
			weldConstraint.Part0 = humanoidRootPart
			weldConstraint.Part1 = fireFliesPart
		end
	end
	
	-- Tweens the moon position
	local tweenInfo = TweenInfo.new(6, Enum.EasingStyle.Linear, Enum.EasingDirection.In)
	local moonGoal = {ClockTime = 0}
	local moonTween = TweenService:Create(Lighting, tweenInfo, moonGoal)
	moonTween:Play()
	
	-- Tweenns the other lighting properties
	module.tweenLighting("BloodMoonEnvironment", tweenInfo)
	

	moonTween.Completed:Connect(function()
		-- When the moon tweeen is completed it hard sets all the other properties that are non tweenable
		Lighting.Brightness = 10
		module.changeLighting("BloodMoonEnvironment")

		-- This runs after delay duration is over and basically reverses everything that happened and brings
		-- happened and brings it to normal (NormalEnnvironmeent)
		task.delay(duration, function()
			local moonGoal2 = {ClockTime = 12}
			local moonTween2 = TweenService:Create(Lighting, tweenInfo, moonGoal2)
			moonTween2:Play()
			module.tweenLighting("NormalEnvironment", tweenInfo)

			-- When moon tweens back to day time, deletes all the firefliespart and weld parts, and changes
			-- the non-tweenable properties of lighting to normal
			moonTween2.Completed:Connect(function()
				for i,v in deleteObjects do
					v:Destroy()
				end
				Lighting.Brightness = 5
				module.changeLighting("NormalEnvironment")
				weatherRunning.Value = false
				weatherBooleans.Blood_Moon.Value = false
			end)
		end)
	end)
end

-- Checks if the object has a mutation
function module.checkMutations(model)
	for i,v in CollectionService:GetTags(model) do
		if moduleData.BiomePriceMultiplier[v] ~= nil then
			return false
		end
	end
	return true
end


-- Runs a 1/256 chance to make the part golden
function module.runGoldenChance(model)
	if math.random(1,256) == 1 and module.checkMutations(model) then -- Checks if other mutations exist & 1/256 chance
		CollectionService:AddTag(model, "Golden")
		for i,v in model:GetDescendants() do
			-- Turns the color property of every part goldenn
			if v:IsA("Part") then
				v.Color = Color3.fromRGB(255, 215, 0)
				module.colorOffsetPart(v, -25, 25)
			end
			
			-- Adds golden particles
			local goldenParticle = particleFolder.GoldenParticles:Clone()
			if v == model.PrimaryPart then
				goldenParticle.Rate = 18
			end
		end
	end
end


-- Runs a 1/1024 chance to make the part rainbow
function module.runRainbowChance(model, plotFolder)
	if math.random(1,1024) == 1 and module.checkMutations(model) then
		CollectionService:AddTag(model, "Rainbow")
		model.Parent = plotFolder.RainbowParts

		-- Adds rainbow particles
		for i,v in model:GetDescendants() do
			local rainbowParticle = particleFolder.RainbowParticles:Clone()
			rainbowParticle.Parent = v
			if v == model.PrimaryPart then
				rainbowParticle.Rate = 18
			end
		end
	end
end


-- Runs a 1/16 chance to make it blood red
function module.runBloodRedChance(model)

		for i,v in model:GetDescendants() do

			end

	end
end

-- Takes the model and finalizes it's position by picking a random point on the passed in grass
function module.finalizePosition(model, grass, scaleFactor, yOffset, rotationMultiplier)
	local scale = module.getExponentialScale(scaleFactor)
	model:PivotTo(CFrame.new(Vector3.new(grass.Position.X + RF.getRandomFloat(-grass.Size.X*0.4, grass.Size.X*0.4), grass.Position.Y + yOffset*scale , grass.Position.Z + RF.getRandomFloat(-grass.Size.Z*0.4, grass.Size.Z*0.4))))
	model:ScaleTo(scale)
	model:SetAttribute("Scale", scale)
end



-- Makes a Savannah leaf on top of the tree with the top position given
function module.makeSavanahLeaf(model, biome, position)
	local referenceLeaf = biomesFolder[biome].TreeLeaf
	
	-- First leaf (bottom one and is a little bit bigger)
	local mainLeaf = referenceLeaf:Clone()
	local mainLeafSize = RF.getRandomFloat(20, 25)
	mainLeaf.Parent = model
	mainLeaf.Size = Vector3.new(mainLeafSize, RF.getRandomFloat(4,5), mainLeafSize)
	mainLeaf.Position = position
	module.colorOffsetPart(mainLeaf, -25, 25)

	-- Second leaf (top one and is a little bit smaller)

	subLeaf.Position = position + Vector3.new(0, mainLeaf.Size.Y/2, 0)
	module.colorOffsetPart(subLeaf, -25, 25)
end

-- Main function for making the map, takes in the plot number (1-4) and a table of coords then makes the map using the given coords
function module.makeMap(plotNumber, tableOfCoords, player)
	-- Variables
	local userId = player.UserId
	local heightScale = RF.getRandomFloat(.022, .026)
	importantValues.HeightScale.Value = heightScale -- Stores it in a intObject if needed later
	local plotFolder = plot["Plot"..plotNumber]
	

	-- Gets a random seed for every biome and saves it to player data
	for biomeName, chanceData in moduleData.BiomeChanceData do
		local biomeSeed
		local savedBiomeSeed = playerData["BiomeSeeds"][player.UserId][biomeName]
		if savedBiomeSeed then -- checks if a seed already exists
			biomeSeed = savedBiomeSeed
		else -- if not sets it
			biomeSeed = math.random(1,100000000)
			playerData["BiomeSeeds"][player.UserId][biomeName] = biomeSeed
		end
	end


	-- Gets a random seed for the height map and cchecks if the coordinate {1,1} is a water chunk and if it is, it gets a new seed
	local heightSeed
	local savedHeightSeed = playerData["HeightSeed"][player.UserId][1]
	if savedHeightSeed then -- checks if a seed already exists
		heightSeed = savedHeightSeed
	else -- if not sets it
		heightSeed = math.random(1,100000000)
		while module.checkHeightSeed(heightScale, heightSeed, plotNumber, {1,1}) == false do
			heightSeed = math.random(1,100000000)
		end
		playerData["HeightSeed"][player.UserId][1] = heightSeed
	end

	-- Loops through the table of coords and adds a chunk for every coord
	for i,v in tableOfCoords do
		module.AddChunk(v[1], v[2], plotNumber, heightSeed, heightScale, false, player)
	end

	-- Gets the buy chunk coordinates then sets their chunk sizes and saves it to player data
	for i,v in RF.getBuyCoordinates(playerData["OwnedChunks"][userId]) do
		if chunkSizes[userId][v[1]..","..v[2]] == nil then
			module.setChunkSize(v[1], v[2], plotNumber, heightSeed, heightScale, userId)
		end
	end
	
	-- Sends the data to the client
	updateClientData:FireClient(player, "ChunkSizes", chunkSizes[userId]) 
	local ownedChunks = playerData["OwnedChunks"][player.UserId]
	updateClientData:FireClient(player, "OwnedChunks", ownedChunks)
end


-- Checks for if the noiselocation is at a certain threshold (smaller than -0.5)
-- This is used to limit the biomes size and get the biome effect
function module.checkHeightSeed(scale, seed, plotNumber, coords)
	local x, z = RF.coordinateScale(increment, plotNumber, coords)

	local noiseLocation = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)
	if noiseLocation < -0.5 or noiseLocation > 0.5 then
		return false
	else
		return true
	end
end

-- Adds a chunk to the given coordinates and seeds
function module.AddChunk(coord1, coord2, plotNumber, heightSeed, heightScale, animation, player)
	-- Variables
	local userId = player.UserId
	local part = chunkBlock:Clone()
	local plotFolder = plot["Plot"..plotNumber]
	local fullCoord = coord1..","..coord2..","..plotNumber
	part.Name = fullCoord

	-- Height noise to get ySize diffferences
	module.makeHeightNoise(coord1, coord2, heightScale, heightSeed, plotNumber, part, userId)

	-- Decides the biome that chunk is gonna be by looping through every biome seed and checking the noise location of the biome seed
	for biomeName, seed in playerData.BiomeSeeds[player.UserId] do
		if CollectionService:HasTag(part, "Water")  then
			-- do nothing
		else
			-- Gets the chance data, scaled x and z annd gets the noise location using those values
			local x, z = RF.coordinateScale(increment, plotNumber, {coord1, coord2})
			local chanceData = moduleData.BiomeChanceData[biomeName]
			local scale = biomeScale * chanceData[2]
			local noiseLocation = math.noise((x + seed) / (increment) * scale, (z + seed) / (increment) * scale)

			-- If validated then replaces that chunk with that seed
			if noiseLocation >= chanceData[1] then
				module.replaceChunk(part, ReplicatedStorage.Biomes[biomeName])
			end
		end
	end

	-- If none of the seeds apply to the chunk then it runs check empty chcunk which basiccally turns it into default grass biome
	module.checkEmptyChunk(part)

	-- This is for if animation should play (when player joins no annimation will be playing but when player buys animationn should play)
	if animation then
		part.Parent = workspace.TempStorage
		table.insert(playerData["OwnedChunks"][userId], {coord1, coord2})
		-- Updates the buy coordinates size info and saves in player data
		for i,v in RF.getBuyCoordinates(playerData["OwnedChunks"][userId]) do 
			if chunkSizes[userId][v[1]..","..v[2]] == nil then
				module.setChunkSize(v[1], v[2], plotNumber, heightSeed, heightScale, userId)
			end
		end
		-- Sends updated data to client and destroys the skeleton part because its not a buy chunk anymore
		updateClientData:FireClient(player, "OwnedChunks", playerData["OwnedChunks"][userId])
		updateClientData:FireClient(player, "ChunkSizes", chunkSizes[userId])
		destroySkeleton:FireClient(player, fullCoord)
		

		local top = part:FindFirstChild("Grass") or part:FindFirstChild("WaterBlock")
		local tweenParts = {top, part}
		local tweenInfo = TweenInfo.new(2, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out, 0, false, 0)

		-- Tweens the grass and the dirt 
		for i,v in ipairs(tweenParts) do
			-- Puts the chunk at the bottom first then tweens it. Makes it start at 1 transparenccy too (invisible)
			local originalPosition = v.Position
			local goal = {Transparency = 0, Position = originalPosition}
			local tween = TweenService:Create(v, tweenInfo, goal)
			v.Position = originalPosition + Vector3.new(0, -20, 0)
			v.Transparency = 1
			tween:Play() 
			part.Parent = plotFolder.ChunkFolder
		end
	end
end


-- Height noise function for the ySize of the chunk and picking which chunks are water
function module.makeHeightNoise(coord1, coord2, scale, seed, plotNumber, part, userId)
	-- Variables
	local x, z = RF.coordinateScale(increment, plotNumber, {coord1, coord2})
	local plotFolder = plot["Plot"..plotNumber]
	local noiseValue = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)
	local normalized = noiseValue + 1           
	local y = normalized * 100
	local yMainPartLocation = y/2
	part:SetAttribute("yValue", y) -- Saves y value as an atrribute of the obejct for later use

	local xSize, ySize, zSize, noiseValue = module.setChunkSize(coord1, coord2, plotNumber, seed, scale, userId) -- Gets the sizes

	-- Applies the sizes annd position
	part.Size = Vector3.new(xSize, y, zSize)
	part.Anchored = true
	part.Position = Vector3.new(x, yMainPartLocation, z)
	part.Parent = plotFolder.ChunkFolder

	-- If statement that if noise value < -.4 then basically make it a water chunk
	if noiseValue < -0.4 then
		CollectionService:AddTag(part, "Water")

		-- Changes the chunk to sand (because under water is sand)
		part.Material = Enum.Material.Sand
		part.Color = Color3.fromRGB(199, 159, 119)
		module.colorOffsetPart(part, -15, 15)

		-- Creates a water block on top of the sand
		local waterBlock = ServerStorage.WaterBlock:Clone()
		waterBlock.Parent = part
		local ySize = 60 - (yMainPartLocation + part.Size.Y/2)
		local yPosition = yMainPartLocation + part.Size.Y/2 + ySize/2
		waterBlock.Size = Vector3.new(part.Size.X, ySize, part.Size.Z)
		waterBlock.Position = Vector3.new(x, yPosition, z)
		module.colorOffsetPart(waterBlock, -15, 15)
	end
end



-- Takes a chunk and changes the biome of that chunk
function module.replaceChunk(part, BiomeType) -- Passes in the biome folder type from ReplicatedStorage.Biomes
	for i,v in part:GetChildren() do
		v:Destroy()
	end

	for i,v in moduleData.GrassTypes do -- Removes all the biome/grass types it has 
		if part:HasTag(v) then
			CollectionService:RemoveTag(part, v)
		end
	end

	CollectionService:AddTag(part, BiomeType.Name) -- Adds the new biome tag

	-- Adds variation to colors and sets material and color of dirt
	part.Color = BiomeType.Dirt.Color
	module.colorOffsetPart(part, -12, 12)
	part.Material = BiomeType.Dirt.Material

	-- Adds variation to colors and sets material of grass and sets position of the grass tto be on top of dirt 
	local GrassClone = BiomeType.Grass:Clone()
	GrassClone.Parent = part
	GrassClone.Size = Vector3.new(part.Size.X, 1, part.Size.Z)
	GrassClone.Position = Vector3.new(part.Position.X, part.Position.Y + part.Size.Y/2 + RF.getRandomFloat(0.45,0.5), part.Position.Z)
	module.colorOffsetPart(GrassClone, -12, 12)
end


-- Checks if the chunk has a biome type and if doesnt it sets as the default normal grass
function module.checkEmptyChunk(chunkPart)
	local hasTag = false
	for x,y in moduleData.GrassTypes do -- Checks if has
		if chunkPart:HasTag(y) then
			hasTag = true
		end
	end
	if hasTag == false then -- If false it sets default grass
		module.replaceChunk(chunkPart, ReplicatedStorage.Biomes.NormalGrass)
		CollectionService:AddTag(chunkPart, "NormalGrass")
	end
end


-- Deletes the map (ran when player leaves)
function module.deleteMap(plotFolder)
	local tableOfPartsFolders = {"ChunkFolder", "RainbowParts", "BuyUIFolder", "SeedDataFolder"}
	for i, v in tableOfPartsFolders do
		for x, y in plotFolder[v]:GetChildren() do
			if y:IsA("Highlight") == false then -- Keeps highlight (used for rainbow parts)
				y:Destroy()
			end
		end
	end
	plotFolder.NumberOfObjects.Value = 0
end

-- Sets a random xSize and zSize and takes  the noise value and sets the ySize for height
-- If the chunk is already saved in player data then it doesnt make a new one just copies it
function module.setChunkSize(coord1, coord2, plotNumber, seed, scale, userId)
	local xSize, ySize, zSize, noiseValue
	local savedSizeTable = chunkSizes[userId][coord1..","..coord2]
	if savedSizeTable then -- Checks if the chunk already exists in player data and if it does it uses its info
		xSize = savedSizeTable[1]
		ySize = savedSizeTable[2]
		zSize = savedSizeTable[3]
		noiseValue = savedSizeTable[4]
	else -- If it doesnt then creases new ones
		local x, z = RF.coordinateScale(increment, plotNumber, {coord1, coord2})
		noiseValue = math.noise((x + seed) / (increment/2.5) * scale, (z + seed) / (increment/2.5) * scale)         
		ySize = (noiseValue + 1 ) * 100

		xSize = increment + RF.getRandomFloat(0, increment)
		zSize = increment + RF.getRandomFloat(0, increment)
		chunkSizes[userId][coord1..","..coord2] = {xSize, ySize, zSize, noiseValue} -- Saves it in player data
	end
	return	xSize, ySize, zSize, noiseValue
end


-- Sets the final HP, price, and current HP of the model
function module.setValueAndHP(model)
	local basePrice = moduleData.ObjectPrice[model.Name]

	-- Gets the mutation multipliers
	local mutationMultiplier = 1
	for i,v in model:GetTags() do
		if v ~= "BreakableModel" then
			mutationMultiplier *= moduleData.mutationMultiplier[v]
		end
	end

	local finalPrice = math.floor(basePrice * mutationMultiplier * RF.getRandomFloat(1,2)) * 3
	local finalHP = math.floor(basePrice* model:GetAttribute("Scale") * RF.getRandomFloat(1,2))

	-- Price
	local valueObject = Instance.new("NumberValue", model)
	valueObject.Name = "Price"
	valueObject.Value = finalPrice

	-- HP
	local HPObject = Instance.new("NumberValue", model)
	HPObject.Name = "HP"
	HPObject.Value = finalHP

	-- CurrentHP
	local currentHP = Instance.new("NumberValue", model)
	currentHP.Name = "CurrentHP"
	currentHP.Value = finalHP
end


-- Given a model it turns it into a tool and puts it in the players backpack (roblox built in stuff)
function module.breakModel(model, player)
	local primaryPart = model.PrimaryPart
	primaryPart.Name = "Handle"
	local tool = Instance.new("Tool", player.Backpack)

	-- Gets the name and adds any mutation to the name if a mutation exists
	local name
	for i,v in CollectionService:GetTags(model) do
		if moduleData.mutationMultiplier[v] ~= nil then
			name = RF.ReplaceUnderscoresWithSpaces(model.Name).." ("..RF.ReplaceUnderscoresWithSpaces(v)..")"
			break
		else
			name = RF.ReplaceUnderscoresWithSpaces(model.Name)
		end
	end
	tool.Name = name
	tool.Grip = CFrame.new(0, 0, primaryPart.Size.Z/2)

	-- Welds everything inside the model together and parents it to the tool then destroys the original model 
	-- Keeps the price of the model so it can be used when player goes to sell it
	for i,v in model:GetDescendants() do
		if v:IsA("Part") then
			v.Parent = tool
			v.Massless = true
			v.Anchored = false
			v.CanCollide = false
			if v.Name == "Handle" then
				continue
			else
				local we  ld = Instance.new("WeldConstraint", tool)
				weld.Part0 = tool.Handle
				weld.Part1 = v
			end
		elseif v.Name == "Price" then
			v.Parent = tool
		end
	end
	model:Destroy()
end


-- Weather manager that gets ran every second from handler
-- Checks if a weather is already running or not, if not then  runs a chance to run each weather then calls those functions
-- And updates the weather booleam and running values
function module.runWeatherManager()
	if weatherRunning.Value == false then
		if math.random(1, weatherChance.Blood_Moon) == 1 then
			weatherRunning.Value = true
			weatherBooleans.Blood_Moon.Value = true
			module.makeBloodMoon(30,40)
		elseif math.random(1, weatherChance.Silver_Night) == 1 then	
			weatherRunning.Value = true
			weatherBooleans.Silver_Night.Value = true
			module.makeSilverNight(math.random(30,40))
		end
	end
end
